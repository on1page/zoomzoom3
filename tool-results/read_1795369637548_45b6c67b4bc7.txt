   152→// località scelta per la nuova scheda: qualsiasi comune, paese o frazione

   154→type ZonePick = {

   156→  province: string | null;

   165→function FindStep({ target, onSelected, onBack }: {
   166→  target: "immobile" | "zona";
   167→  onSelected: (ent: SelEntity) => void;
   168→  onBack: () => void;
   169→}) {
   170→  const { boot, toast } = useApp();
   171→  const [q, setQ] = useState("");
   172→  const [rows, setRows] = useState<SearchRow[] | null>(null);
   173→  const [busy, setBusy] = useState(false);
   174→
   175→  // stato "aggiungi immobile"
   176→  const [adding, setAdding] = useState(false);
   177→  const [address, setAddress] = useState("");
   178→  const [zone, setZone] = useState<ZonePick | null>(null);
   179→  // conferma esplicita della città scelta (tasto “Conferma” accanto a
   180→  // “Cambia”): la città è già valida dopo la scelta, ma così chi compila
   181→  // vede CHIARAMENTE che il valore è memorizzato — come per l'indirizzo
   182→  const [zoneOk, setZoneOk] = useState(false);
   183→  const [addBusy, setAddBusy] = useState(false);
   184→  const [geo, setGeo] = useState<LatLng | null>(null);
   185→  const [geoQ, setGeoQ] = useState("");
   186→  const [geoRes, setGeoRes] = useState<Array<{ label: string; lat: number; lng: number; zoneName: string | null; province: string | null; region: string | null; road?: string | null; houseNumber?: string | null }> | null>(null);
   187→  const [geoBusy, setGeoBusy] = useState(false);
   188→  const [gpsStage, setGpsStage] = useState<"idle" | "gps" | "addr">("idle");
   189→  const [addErr, setAddErr] = useState<string | null>(null);
   190→
   191→  // zona: elenco delle città già sul catalogo (dalla mappa pubblica)
   192→  const [pubZones, setPubZones] = useState<PubZone[] | null>(null);
   193→  const [zoneBusy, setZoneBusy] = useState(false);
   194→
   195→  // ogni via che imposta la città passa da qui: la conferma precedente
   196→  // (eventuale) si azzera finché non si conferma la nuova
   197→  const applyZone = (z: ZonePick | null) => {
   198→    setZone(z);
   199→    setZoneOk(false);
   200→  };
   201→
   202→  const search = useCallback(async (query: string) => {
   203→    setBusy(true);
   204→    const res = await api<{ rows: SearchRow[] }>(`/api/public/search?cat=immobile${query ? `&q=${encodeURIComponent(query)}` : ""}`);
   205→    setRows(res.ok ? res.rows : []);
   206→    setBusy(false);
   207→  }, []);
   208→
   209→  useEffect(() => {
   210→    if (target === "immobile") void search("");
   211→    else void (async () => {
   212→      const res = await api<{ zones: PubZone[] }>("/api/public/map");
   213→      setPubZones(res.ok ? res.zones : []);
   214→    })();
   215→  }, [target, search]);
   216→
   217→  // attiva la scheda recensibile di una città: se esiste già si usa quella,
   218→  // altrimenti nasce adesso (copertura nazionale, paesi inclusi)
   219→  const activateZone = async (z: { id?: string; name: string; lat: number; lng: number; province?: string | null; region?: string | null; entityId?: string | null }) => {
   220→    if (z.entityId) { onSelected({ id: z.entityId, name: z.name, categoryId: "zona" }); return; }
   221→    setZoneBusy(true);
   222→    const res = await api<{ entityId: string; name: string; created: boolean }>("/api/zones", {
   223→      body: { name: z.name, province: z.province ?? null, region: z.region ?? null, lat: z.lat, lng: z.lng },
   224→    });
   225→    setZoneBusy(false);
   226→    if (res.ok && res.entityId) {
   227→      toast(res.created ? `Scheda di ${z.name} attivata: ora scrivi la recensione.` : `Scheda di ${z.name} già pronta: ora scrivi la recensione.`);
   228→      onSelected({ id: res.entityId, name: z.name, categoryId: "zona" });
   229→    } else {
   230→      toast("Non riesco ad attivare la scheda della città: riprova tra poco.", "err");
   231→    }
   232→  };
   233→
   234→  const doGeocode = async () => {
   235→    if (geoQ.trim().length < 3) return;
   236→    setGeoBusy(true); setGeoRes(null);
   237→    const res = await api<{ results: Array<{ label: string; lat: number; lng: number; zoneName: string | null; province: string | null; region: string | null; road?: string | null; houseNumber?: string | null }> }>(`/api/public/geocode?q=${encodeURIComponent(geoQ.trim())}`);
   238→    setGeoRes(res.ok ? res.results : []);
   239→    setGeoBusy(false);
   240→  };
   241→
   242→  // GPS + reverse geocoding: chi è davanti all'immobile preme un bottone e
   243→  // il modulo si riempie da solo (indirizzo reale, città, pin). Le coordinate
   244→  // vanno al geocoder pubblico SOLO in questo momento, su richiesta esplicita.
   245→  const useGpsAddress = () => {
   246→    if (!navigator.geolocation) { toast("Questo browser non espone la posizione: usa la ricerca o la mappa.", "err"); return; }
   247→    setGpsStage("gps");
   248→    navigator.geolocation.getCurrentPosition(
   249→      async pos => {
   250→        const pt = { lat: Math.round(pos.coords.latitude * 1e6) / 1e6, lng: Math.round(pos.coords.longitude * 1e6) / 1e6 };
   251→        setGeo(pt);
   252→        setGpsStage("addr");
   253→        const res = await api<{ found: boolean; address: { road: string | null; houseNumber: string | null; zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${pt.lat}&lng=${pt.lng}`);
   254→        setGpsStage("idle");
   255→        if (res.ok && res.found && res.address) {
   256→          const a = res.address;
   257→          const street = [a.road, a.houseNumber].filter(Boolean).join(" ");
   258→          if (street) setAddress(street);
   259→          if (a.zoneName) applyZone({ name: a.zoneName, province: a.province, region: a.region, lat: pt.lat, lng: pt.lng });
   260→          toast("Indirizzo riconosciuto: controlla e correggi se serve.");
   261→        } else {
   262→          toast("Posizione rilevata, ma l'indirizzo non risulta mappato: inseriscilo a mano (il pin è già a posto).", "err");
   263→        }
   264→      },
   265→      () => { setGpsStage("idle"); toast("Posizione non disponibile: consenti l'accesso dal browser oppure cerca a mano.", "err"); },
   266→      { timeout: 9000, maximumAge: 30000, enableHighAccuracy: true },
   267→    );
   268→  };
   269→
   270→  // conferma di un INDIRIZZO dai suggerimenti del campo indirizzo: riempie
   271→  // via/civico, città e pin in un colpo solo — l'utente vede esattamente
   272→  // cosa sta per salvare e lo conferma scegliendo il suggerimento
   273→  const pickAddress = (p: AddressPlace) => {
   274→    setGeo({ lat: p.lat, lng: p.lng });
   275→    if (p.zoneName) {
   276→      applyZone({ name: p.zoneName, province: p.province, region: p.region, lat: p.lat, lng: p.lng });
   277→    }
   278→  };
   279→
   280→  // conferma di un risultato della ricerca sulla mappa: piazza il pin e
   281→  // ricava la città. Se il risultato non porta il comune (es. una via di
   282→  // campagna mappata senza riferimento), lo chiede al reverse geocoder sul
   283→  // punto scelto: un'azione esplicita dell'utente, come il GPS. Prima di
   284→  // questa correzione il pin veniva piazzato ma la città restava vuota,
   285→  // senza spiegazioni: la località trovata non era confermabile.
   286→  const pickGeoResult = async (g: NonNullable<typeof geoRes>[number]) => {
   287→    setGeo({ lat: g.lat, lng: g.lng });
   288→    setGeoRes(null);
   289→    const street = [g.road, g.houseNumber].filter(Boolean).join(" ");
   290→    if (street && !address.trim()) setAddress(street);
   291→    if (zone) return; // città già scelta: il risultato serve solo per il pin
   292→    if (g.zoneName) {
   293→      applyZone({ name: g.zoneName, province: g.province, region: g.region, lat: g.lat, lng: g.lng });
   294→      return;
   295→    }
   296→    const res = await api<{ found: boolean; address: { zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${g.lat}&lng=${g.lng}`);
   297→    if (res.ok && res.found && res.address?.zoneName) {
   298→      applyZone({ name: res.address.zoneName, province: res.address.province, region: res.address.region, lat: g.lat, lng: g.lng });
   299→    } else {
   300→      toast("Posizione segnata sulla mappa: scegli anche la città nel campo “Città, comune o paese”.", "err");
   301→    }
   302→  };
   303→
   304→  const addEntity = async () => {
   305→    if (addBusy) return;
   306→    setAddErr(null);
   307→    setAddBusy(true);
   308→    const res = await api<{ entity: { id: string; name: string } }>("/api/entities", {
   309→      body: {
   310→        address: address.trim(),
   311→        // città già presente → id diretto; nuova località → nome+coordinate.
   312→        // Le coordinate vanno solo se davvero valide (numeri finiti): il
   313→        // server ha comunque i fallback (zona nota col nome, pin sulla mappa)
   314→        zoneId: zone?.localId,
   315→        zone: zone && !zone.localId
   316→          ? {
   317→            name: zone.name, province: zone.province, region: zone.region,
   318→            lat: Number.isFinite(zone.lat) ? zone.lat : undefined,
   319→            lng: Number.isFinite(zone.lng) ? zone.lng : undefined,
   320→          }
   321→          : undefined,
   322→        geo,
   323→      },
   324→    });
   325→    setAddBusy(false);
   326→    if (res.ok) {
   327→      if (res.entity) {
   328→        toast("Immobile aggiunto: ora scrivi la recensione.");
   329→        onSelected({ id: res.entity.id, name: res.entity.name, categoryId: "immobile" });
   330→      }
   331→      return;
   332→    }
   333→    const ex = (res as { existingId?: string }).existingId;
   334→    if (ex) {
   335→      toast("Questo immobile è già presente: puoi recensire la scheda esistente.");
   336→      onSelected({ id: ex, name: address.trim(), categoryId: "immobile" });
   337→      return;
   338→    }
   339→    setAddErr(res.error);
   340→  };
   341→
   342→  const zoneName = (id: string | null) => boot?.zones.find(z => z.id === id)?.name || "";
   343→
   344→  return (
   345→    <div className="container" style={{ maxWidth: 860, paddingTop: 26 }}>
   346→      <div className="section-head">
   347→        <div>
   348→          <h1 style={{ margin: 0 }}>{target === "immobile" ? "Quale immobile?" : "Quale città?"}</h1>
   349→          <p className="sub">{target === "immobile" ? "Cercalo tra le schede esistenti oppure aggiungilo dall'indirizzo: basta quello." : "Una città già attiva, oppure qualsiasi comune o paese d'Italia: la scheda si attiva al momento."}</p>
   350→        </div>
   351→        <button className="btn btn-ghost btn-sm" onClick={onBack}>← Cambia categoria</button>
   352→      </div>
   353→
   354→      {target === "zona" ? (
   355→        <>
   356→          {pubZones == null ? (
   357→            <p className="muted">Carico le città…</p>
   358→          ) : (
   359→            <div className="grid grid-3">
   360→              {pubZones.map(z => (
   361→                <button key={z.id} className="card card-click" style={{ textAlign: "left" }} disabled={zoneBusy}
   362→                  onClick={() => void activateZone({ id: z.id, name: z.name, lat: z.lat, lng: z.lng, entityId: z.entityId })}>
   363→                  <h3 style={{ margin: 0 }}>{z.name}</h3>
   364→                  <span className="small muted">{z.entityId ? "Scheda attiva · servizi, trasporti, sicurezza, costi" : "Sul catalogo · scheda da attivare"}</span>
   365→                </button>
   366→              ))}
   367→            </div>
   368→          )}
   369→          <div className="card mt3">
   370→            <h2 style={{ marginBottom: 4 }}>La tua città non è tra queste?</h2>
   371→            <p className="small muted" style={{ marginTop: 0 }}>Cerca qualsiasi comune, paese o frazione d&apos;Italia: la scheda nasce ora e resta nel catalogo.</p>
   372→            <PlaceSearch id="zps-city" localZones={boot?.zones}
   373→              placeholder="Es. Corte de' Frati, Marciaga, Alassio…"
   374→              onPick={p => void activateZone({
   375→                name: p.zoneName || p.name,
   376→                province: p.province, region: p.region,
   377→                lat: p.lat, lng: p.lng,
   378→                entityId: p.localId ? (pubZones?.find(z => z.id === p.localId)?.entityId ?? null) : null,
   379→              })} />
   380→            {zoneBusy ? <p className="small muted mb0">Attivo la scheda…</p> : null}
   381→          </div>
   382→        </>
   383→      ) : (
   384→        <>
   385→          <form className="searchbar" role="search" onSubmit={e => { e.preventDefault(); void search(q.trim()); }}>

   494→          </div>

   497→    </div>

   502→function OpenReviewForm({ sel, onBack, onDone }: {
   503→  sel: SelEntity;
   504→  onBack: () => void;
   505→  onDone: (entityId: string, premiumDays: number) => void | Promise<void>;
   506→}) {
   507→  const { boot, toast } = useApp();
   508→  const cat = boot?.categories.find(c => c.id === sel.categoryId) || null;
   509→  const [kind, setKind] = useState("");
   510→  const [form, setForm] = useState<FormState>({
   511→    kind: "", score: 5, crit: {} as Record<string, number>,
   512→    priceAmount: "", priceUnit: "", priceKind: "Concluso/pagato",
   513→    pros: "", cons: "", text: "", honesty: false, hp: "",

   517→  const [busy, setBusy] = useState(false);
   518→
   519→  useEffect(() => { if (cat?.experienceKinds.length) setKind(cat.experienceKinds[0]); }, [cat]);
   520→

   522→

   524→    <div className="container" style={{ maxWidth: 760, paddingTop: 26, paddingBottom: 30 }}>
   525→      <div className="section-head">
   526→        <div>
   527→          <h1 style={{ margin: 0 }}>Scrivi la recensione</h1>

   533→      </div>
   534→      <Alert kind="info">
   535→        Stai scrivendo senza invito: la recensione sarà marcata <b>“esperienza dichiarata”</b> (non “verificata”). Restano valide le regole: segnalazioni, moderazione con decisioni motivate e revoca del Premium in caso di frode accertata.
   536→      </Alert>

   538→        <Alert kind="ok"><b>Membership Premium.</b> Pubblicando questa recensione ottieni <b>{boot.settings.premiumDaysPerReview} giorni di Premium</b>: leggi le recensioni integrali di tutte le schede.
