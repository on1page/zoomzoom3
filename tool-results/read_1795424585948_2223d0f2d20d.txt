    88→function esc(s: string): string {
    89→  return s.replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" })[c] as string);
    90→}

    92→/** Cerchio geodetico approssimato (WGS84) intorno a un centro: serve per
    93→ *  l'area indicativa della città, come faceva L.circle di Leaflet. */
    94→function circlePolygon(lat: number, lng: number, radiusM: number, segments = 48): GeoJSON.Polygon {
    95→  const dLat = radiusM / 111_320;
    96→  const dLng = radiusM / (111_320 * Math.max(0.2, Math.cos((lat * Math.PI) / 180)));
    97→  const ring: Array<[number, number]> = [];
    98→  for (let i = 0; i <= segments; i++) {
    99→    const a = (i / segments) * 2 * Math.PI;
   100→    ring.push([lng + dLng * Math.cos(a), lat + dLat * Math.sin(a)]);
   101→  }
   102→  return { type: "Polygon", coordinates: [ring] };
   103→}

   143→  // regioni: chiave = nome normalizzato (proprietà "n" del GeoJSON ISTAT)
   144→  function regionMatch(): MapLibreNS.ExpressionSpecification | string {
   145→    const d = live.current.data;
   146→    const byNorm = new Map<string, string>();
   147→    for (const r of d?.admin.regions || []) {
   148→      const c = regionCanonical(r.name);
   149→      if (c) byNorm.set(normName(c), scoreBandColor(r.mean));
   150→    }
   151→    const pairs: Array<string> = [];
   152→    for (const f of regioniRef.current?.features || []) {
   153→      const n = String((f.properties as { n?: string } | null)?.n || "");
   154→      const col = byNorm.get(n);
   155→      if (n && col) pairs.push(n, col);
   156→    }
   157→    // match senza coppie non è un'espressione valida: si usa il colore neutro
   158→    return pairs.length
   159→      ? ["match", ["get", "n"], ...pairs, "rgba(0,0,0,0)"] as unknown as MapLibreNS.ExpressionSpecification
   160→      : "rgba(0,0,0,0)";

   397→            fillOp: z.aggregate ? 0.14 : 0.05,
   398→            lineColor: isActive ? "#33658A" : color,
   399→            lineW: isActive ? 2.6 : isMine ? 2.1 : 1.2,
   400→            lineOp: isActive || isMine ? 0.9 : 0.5,
   401→          },
   402→          geometry: circlePolygon(z.lat, z.lng, z.radiusM),
   403→        };
   404→      }),
   405→    };
   406→    (map.getSource("zz-zone-areas") as MapLibreNS.GeoJSONSource | undefined)?.setData(areas);
   407→
   408→    // punti immobile (solo a zoom ravvicinato, modalità immobili)
   409→    const propsFc: GeoJSON.FeatureCollection = {
   410→      type: "FeatureCollection",
   411→      features: showProps ? d.markers.map(mk => {
   412→        const city = d.zones.find(z => z.id === mk.zoneId)?.name || "";
   413→        return {
   414→          type: "Feature",
   415→          properties: { id: mk.id, name: mk.name, mean: mk.mean, n: mk.n, isDemo: mk.isDemo, city },
   416→          geometry: { type: "Point", coordinates: [mk.lng, mk.lat] },
   417→        };
   418→      }) : [],
   419→    };
   420→    (map.getSource("zz-props") as MapLibreNS.GeoJSONSource | undefined)?.setData(propsFc);
   421→  }, []);
   422→
   423→  /* ---- comuni: contorni on demand quando si zooma/cerca ---- */
   424→  const fetchComuni = useCallback(async () => {
   425→    const map = mapRef.current;
   426→    const d = live.current.data;
   427→    if (!map || !d || map.getZoom() < Z_COMUNE_FETCH) return;
   428→    // margine intorno alla vista (~15%): si chiedono i comuni visibili o in
   429→    // arrivo, senza il metodo .pad() che LngLatBounds non ha
   430→    const b = map.getBounds();
   431→    const latSpan = Math.max(0.02, b.getNorth() - b.getSouth());
   432→    const lngSpan = Math.max(0.02, b.getEast() - b.getWest());
   433→    const margin = (v: number) => v * 0.15;
   434→    const inView = (lng: number, lat: number) =>
   435→      lng >= b.getWest() - margin(lngSpan) && lng <= b.getEast() + margin(lngSpan) &&
   436→      lat >= b.getSouth() - margin(latSpan) && lat <= b.getNorth() + margin(latSpan);
   437→    const pending = d.zones
   438→      .filter(z => z.place && !comuniAskedRef.current.has(z.id))
   439→      .filter(z => inView(z.lng, z.lat))
   440→      .slice(0, 6);
   441→    for (const z of pending) {
   442→      comuniAskedRef.current.add(z.id);
   443→      const q = new URLSearchParams({ name: z.name, province: z.province || "", region: z.region || "" });
   444→      const res = await api<{ polygon: { type: string; coordinates: unknown } | null }>(`/api/public/zone-outline?${q.toString()}`);
   445→      if (res.ok && res.polygon) {
   446→        comuniFeatRef.current.set(z.id, {
   447→          type: "Feature",
   448→          properties: { id: z.id, nome: z.name, n: normName(z.name), mean: z.place!.mean, nrev: z.place!.n, entityId: z.entityId },
   449→          geometry: res.polygon as GeoJSON.Geometry,
   450→        });
   451→      }
   452→    }
   453→    if (pending.length) {
   454→      (map.getSource("zz-comuni") as MapLibreNS.GeoJSONSource | undefined)
   455→        ?.setData({ type: "FeatureCollection", features: [...comuniFeatRef.current.values()] });
   456→      // il match dei colori cambia solo se arrivano nuovi comuni
   457→      if (map.getLayer("zz-comuni-fill")) {
   458→        map.setPaintProperty("zz-comuni-fill", "fill-color", comuneMatch());
   459→      }
   460→    }
   461→  }, []);
   462→
   463→  // inizializzazione MapLibre (una sola volta, all'arrivo dei dati)
   464→  useEffect(() => {
   465→    if (!data || !holderRef.current || mapRef.current) return;
   466→    const holder = holderRef.current;
   467→    let disposed = false;
   468→    let ro: ResizeObserver | null = null;
   469→    let roFit: ResizeObserver | null = null;
   470→    void (async () => {
   471→      const maplibregl = await import("maplibre-gl");
   472→      if (disposed || !holderRef.current) return;
   473→      MLRef.current = maplibregl as unknown as typeof MapLibreNS;
   474→      setupMapLibre(MLRef.current);
   475→
   476→      // base: stile MapTiler OMT minimale se c'è la chiave (verificato con
   477→      // una richiesta: chiave errata o stile assente → ripiego OSM raster)
   478→      let style: string | MapLibreNS.StyleSpecification = osmRasterStyle();
   479→      if (MAPTILER_KEY) {
   480→        try {
   481→          const ctrl = new AbortController();
   482→          const timer = setTimeout(() => ctrl.abort(), 6000);
   483→          const r = await fetch(MAPTILER_STYLE_URL, { signal: ctrl.signal });
   484→          clearTimeout(timer);
   485→          if (!r.ok) throw new Error("stile MapTiler non disponibile");
   486→          style = (await r.json()) as MapLibreNS.StyleSpecification;
   487→        } catch {
   488→          // nessun problema: la mappa funziona con i tile pubblici OSM
   489→        }
   490→      }
   491→
   492→      const map = new maplibregl.Map({
   493→        container: holder,
   494→        style,
   495→        // vista di ripiego già sull'Italia: se il contenitore non fosse
   496→        // ancora dimensionato, la mappa non finirebbe mai altrove
   497→        center: [12.5, 42.5],
   498→        zoom: 4.6,
   499→        attributionControl: false,
   500→        maxZoom: 19,
   501→        minZoom: 3.6,
   502→        dragRotate: false,
   503→        pitchWithRotate: false,
   504→        touchPitch: false,
   505→      });
   506→      mapRef.current = map;
   507→      map.addControl(new maplibregl.NavigationControl({ visualizePitch: false }), "top-right");
   508→      map.addControl(new maplibregl.ScaleControl({ unit: "metric", maxWidth: 110 }), "bottom-left");
   509→      map.addControl(new maplibregl.AttributionControl({
   510→        compact: true,
   511→        customAttribution: MAPTILER_KEY
   512→          ? '<a href="https://www.maptiler.com/copyright/" target="_blank" rel="noopener">© MapTiler</a> <a href="https://www.openstreetmap.org/copyright" target="_blank" rel="noopener">© OpenStreetMap contributors</a>'
   513→          : undefined,
   514→      }));
   515→      // inquadra l'Italia quando la mappa è davvero pronta (contenitore
   516→      // dimensionato e stile caricato): un fitBounds troppo presto, con il
   517→      // contenitore ancora a zero pixel, porterebbe la camera altrove
   518→      let fitted = false;
   519→      const fitItaly = () => {
   520→        if (fitted || !data) return;
   521→        fitted = true;
   522→        const initial = activeZone || myZone;
   523→        if (initial && data.zones.some(z => z.id === initial)) {
   524→          const z = data.zones.find(zz => zz.id === initial)!;
   525→          map.jumpTo({ center: [z.lng, z.lat], zoom: 12 });
   526→        } else {
   527→          map.fitBounds(ITALY, { padding: 14 });
   528→        }
   529→      };
   530→      map.once("load", () => fitItaly());
   531→      // se il contenitore cambia dimensione prima del load (layout che si
   532→      // assesta), l'inquadratura dell'Italia resta quella giusta
   533→      const roFitLocal = new ResizeObserver(() => { if (!fitted && map.isStyleLoaded()) fitItaly(); });
   534→      roFit = roFitLocal;
   535→      roFitLocal.observe(holder);
   536→      map.on("resize", () => { if (!fitted && map.isStyleLoaded()) fitItaly(); });
   537→
   538→      // confini ISTAT delle regioni: GeoJSON locale semplificato (92 KB)
   539→      void fetch("/regioni-italia.geojson").then(r => (r.ok ? r.json() : null)).then((gj: GeoJSON.FeatureCollection | null) => {
   540→        if (!gj || disposed) return;
   541→        regioniRef.current = {
   542→          type: "FeatureCollection",
   543→          features: gj.features.map(f => ({
   544→            type: "Feature",
   545→            properties: { nome: (f.properties as { nome?: string } | null)?.nome || "", n: normName((f.properties as { nome?: string } | null)?.nome) },
   546→            geometry: f.geometry,
   547→          })),
   548→        };
   549→        (map.getSource("zz-regioni") as MapLibreNS.GeoJSONSource | undefined)?.setData(regioniRef.current);
   550→        if (map.getLayer("zz-regioni-fill")) {
   551→          map.setPaintProperty("zz-regioni-fill", "fill-color", regionMatch());
   552→        }
   553→      });
   554→
   555→      map.on("style.load", () => { installLayers(); redraw(); });
   556→      map.on("zoomend", () => { redraw(); });
   557→      map.on("moveend", () => { redraw(); void fetchComuni(); });
   558→
   559→      // popup dei punti immobile
   560→      map.on("click", "zz-props", e => {
   561→        const f = e.features?.[0];
   562→        if (!f) return;
   563→        const p = f.properties as { name: string; city: string; mean: number | null; n: number; isDemo: boolean; id: string };
   564→        const mean = p.mean == null ? null : Number(p.mean);
   565→        new maplibregl.Popup({ className: "zz-popup", closeButton: true, maxWidth: "280px" })
   566→          .setLngLat(e.lngLat)
   567→          .setHTML(
   568→            `<div class="zz-pop">
   569→              <strong>${esc(p.name)}</strong>${p.isDemo ? ' <span class="badge-demo">Dato fittizio</span>' : ""}
   570→              <div class="zz-pop-sub">Immobile${p.city ? ` · ${esc(p.city)}` : ""}</div>
   571→              <div class="zz-pop-score" style="color:${scoreBandColor(mean)}">
   572→                ${mean == null ? "Nessuna recensione" : `${fmt1(mean)}/10 · ${p.n} ${p.n === 1 ? "recensione" : "recensioni"}`}
   573→              </div>
   574→              <a href="#/scheda/${esc(p.id)}" class="zz-pop-link">Apri la scheda</a>
   575→            </div>`)
   576→          .addTo(map);
   577→      });
   578→      // popup del comune (livello comunale)
   579→      map.on("click", "zz-comuni-fill", e => {
   580→        const f = e.features?.[0];
   581→        if (!f) return;
   582→        const p = f.properties as { nome: string; mean: number; nrev: number; entityId: string | null };
   583→        new maplibregl.Popup({ className: "zz-popup", closeButton: true, maxWidth: "280px" })
   584→          .setLngLat(e.lngLat)
   585→          .setHTML(
   586→            `<div class="zz-pop">
   587→              <strong>${esc(p.nome)}</strong>
   588→              <div class="zz-pop-sub">Comune · reputazione del territorio</div>
   589→              <div class="zz-pop-score" style="color:${scoreBandColor(p.mean)}">
   590→                ${fmt1(p.mean)}/10 · ${p.nrev} ${p.nrev === 1 ? "recensione" : "recensioni"}
   591→              </div>
   592→              ${p.entityId ? `<a href="#/zona/${esc(p.entityId)}" class="zz-pop-link">Apri la scheda della città</a>` : ""}
   593→            </div>`)
   594→          .addTo(map);
   595→      });
   596→      for (const layerId of ["zz-props", "zz-comuni-fill"]) {
   597→        map.on("mouseenter", layerId, () => { map.getCanvas().style.cursor = "pointer"; });
   598→        map.on("mouseleave", layerId, () => { map.getCanvas().style.cursor = ""; });
   599→      }
   600→
   601→      installLayers();
   602→      redraw();
   603→      void fetchComuni();
   604→      // tiene la mappa allineata quando il contenitore cambia dimensione
   605→      ro = new ResizeObserver(() => map.resize());
   606→      ro.observe(holder);
   607→    })();
   608→    return () => {
   609→      disposed = true;
   610→      ro?.disconnect();
   611→      roFit?.disconnect();
   612→      for (const m of markersRef.current) m.remove();
   613→      markersRef.current = [];
   614→      mapRef.current?.remove();
   615→      mapRef.current = null;
   616→      MLRef.current = null;
   617→      comuniAskedRef.current = new Set();
   618→      comuniFeatRef.current = new Map();
   619→    };
   620→  }, [data]);
   621→
   622→  // ridisegno quando cambiano dati, modalità, zona attiva o città di riferimento
   623→  useEffect(() => {
   624→    live.current = { data, mode, activeZone, myZone };
   625→    const map = mapRef.current;
   626→    if (!map) return;
   627→    if (map.isStyleLoaded() && map.getStyle()) {
   628→      installLayers();
   629→      redraw();
   630→    }
   631→  }, [data, mode, activeZone, myZone, installLayers, redraw]);
   632→
   633→  const focusZone = useCallback((zoneId: string, animate = true) => {
   634→    const map = mapRef.current;
   635→    const z = live.current.data?.zones.find(zz => zz.id === zoneId);
   636→    if (!map || !z) return;
   637→    if (animate) map.flyTo({ center: [z.lng, z.lat], zoom: 12, duration: 0.9 });
   638→    else map.jumpTo({ center: [z.lng, z.lat], zoom: 12 });
   639→  }, []);
   640→  const italyView = useCallback(() => {
   641→    mapRef.current?.fitBounds(ITALY, { padding: 14 });
   642→  }, []);
   643→
   644→  const useGps = () => {
   645→    if (!navigator.geolocation || !data) {
   646→      toast("Geolocalizzazione non disponibile in questo browser: scegli la città dall'elenco.", "err");
   647→      return;
   648→    }
   649→    toast("Rilevo la posizione… le coordinate servono solo al momento e non vengono salvate.");
   650→    navigator.geolocation.getCurrentPosition(
   651→      pos => {
   652→        // città coperta più vicina (distanza locale approssimata)
   653→        const R = 6371, rad = Math.PI / 180;
   654→        const m = Math.cos(pos.coords.latitude * rad);
   655→        let best: MapData["zones"][number] | null = null, bd = Infinity;
   656→        for (const z of data.zones) {
   657→          const dLat = (z.lat - pos.coords.latitude) * rad;
   658→          const dLng = (z.lng - pos.coords.longitude) * rad;
   659→          const d = R * Math.sqrt(dLat * dLat + (dLng * m) * (dLng * m));
   660→          if (d < bd) { bd = d; best = z; }
   661→        }
   662→        if (best) {
   663→          setMyZone(best.id);
   664→          focusZone(best.id, true);
   665→          toast(`Città di riferimento: ${best.name}. Coordinate scartate.`);
   666→        }
   667→      },
   668→      err => {
   669→        const why = err.code === 1 ? "autorizzazione negata" : err.code === 2 ? "posizione non disponibile" : "tempo scaduto";
   670→        toast(`GPS non utilizzabile (${why}): scegli la città dall'elenco.`, "err");
   671→      },
   672→      { timeout: 8000, maximumAge: 600000, enableHighAccuracy: false }
   673→    );
   674→  };
   675→
   676→  if (!data) {
   677→    return (
   678→      <div className="map-viewport" role="application" aria-label="Mappa in caricamento">
   679→        <p className="map-loading">Caricamento della mappa…</p>
   680→      </div>
   681→    );
   682→  }
   683→
   684→  return (
   685→    <div className="map-viewport" role="application"
   686→      aria-label="Mappa delle reputazioni del territorio italiano (cartografia MapTiler/OpenStreetMap). Trascina per esplorare, usa la rotella o i pulsanti per lo zoom. A livello nazionale le regioni sono colorate per reputazione; zoomando passano province e comuni. I pin delle città aprono la loro scheda, i punti immobile aprono una scheda sintetica.">
   687→      <div className="map-holder" ref={holderRef}></div>
   688→      <div className="map-loc">
   689→        <span aria-hidden="true">◎</span>
   690→        <select className="map-zone-select" aria-label="La tua città di riferimento"
   691→          value={myZone || ""}
   692→          onChange={e => {
   693→            const zid = e.target.value || null;
   694→            setMyZone(zid);
   695→            if (zid) {
   696→              focusZone(zid, true);
   697→              toast("Città di riferimento: " + (data.zones.find(z => z.id === zid)?.name || "") + ". Preferenza salvata solo su questo dispositivo.");
   698→            } else {
   699→              italyView();
   700→              toast("Città di riferimento rimossa.");
   701→            }
   702→          }}>
   703→          <option value="">La tua città…</option>
   704→          {data.zones.map(z => <option key={z.id} value={z.id}>{z.name}</option>)}
   705→        </select>
   706→        <button className="btn btn-ghost btn-xs" onClick={useGps}
   707→          title="Usa la mia posizione: le coordinate servono solo al momento e non vengono salvate">GPS</button>
   708→      </div>
   709→      <div className="map-italy-btn">
   710→        <button type="button" onClick={italyView} title="Torna alla vista dell'Italia intera: regioni colorate per reputazione">Vista Italia</button>
   711→      </div>
   712→    </div>
   713→  );
   714→}
