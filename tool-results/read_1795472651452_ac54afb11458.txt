    99→/** Cerchio geodetico approssimato (WGS84) intorno a un centro: serve per
   100→ *  l'area indicativa della città, come faceva L.circle di Leaflet. */
   101→function circlePolygon(lat: number, lng: number, radiusM: number, segments = 48): GeoJSON.Polygon {
   102→  const dLat = radiusM / 111_320;
   103→  const dLng = radiusM / (111_320 * Math.max(0.2, Math.cos((lat * Math.PI) / 180)));
   104→  const ring: Array<[number, number]> = [];
   105→  for (let i = 0; i <= segments; i++) {
   106→    const a = (i / segments) * 2 * Math.PI;
   107→    ring.push([lng + dLng * Math.cos(a), lat + dLat * Math.sin(a)]);
   108→  }
   109→  return { type: "Polygon", coordinates: [ring] };
   110→}

   170→  // regioni: chiave = nome normalizzato (proprietà "n" del GeoJSON ISTAT)
   171→  function regionMatch(): MapLibreNS.ExpressionSpecification | string {
   172→    const d = live.current.data;
   173→    const byNorm = new Map<string, AdminRow>();
   174→    for (const r of d?.admin.regions || []) {
   175→      const c = regionCanonical(r.name);
   176→      if (c) byNorm.set(normName(c), r);
   177→    }
   178→    const pairs: Array<string> = [];
   179→    for (const f of regioniRef.current?.features || []) {
   180→      const n = String((f.properties as { n?: string } | null)?.n || "");
   181→      const row = n ? byNorm.get(n) : undefined;
   182→      const v = row ? rowValue(row) : null;
   183→      if (n && v != null) pairs.push(n, scoreBandColor(v));
   184→    }
   185→    // match senza coppie non è un'espressione valida: si usa il colore neutro
   186→    return pairs.length
   187→      ? ["match", ["get", "n"], ...pairs, "rgba(0,0,0,0)"] as unknown as MapLibreNS.ExpressionSpecification
   188→      : "rgba(0,0,0,0)";

   193→  function provinceMatch(): MapLibreNS.ExpressionSpecification | string {

   207→      : "rgba(0,0,0,0)";
