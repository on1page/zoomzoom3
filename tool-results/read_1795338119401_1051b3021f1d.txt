   158→function ApplyForm({ onDone }: { onDone: () => void }) {
   159→  const { toast } = useApp();
   160→  const [f, setF] = useState({
   161→    firstName: "", lastName: "", profession: "", title: "", bio: "",
   162→    radiusKm: 30, feeEuros: "", vatNumber: "", iban: "", phone: "",
   163→  });
   164→  const [zone, setZone] = useState<Place | null>(null);
   165→  const [photo, setPhoto] = useState<string | null>(null);
   166→  const [docsQ, setDocsQ] = useState<DocDraft[]>([]);
   167→  const [docsD, setDocsD] = useState<DocDraft[]>([]);
   168→  const [docsE, setDocsE] = useState<DocDraft[]>([]);
   169→  const [err, setErr] = useState<string | null>(null);
   170→  const [busy, setBusy] = useState(false);
   171→
   172→  const pickFiles = async (files: FileList | null, kind: string, set: (d: DocDraft[]) => void, current: DocDraft[]) => {
   173→    if (!files || !files.length) return;
   174→    const drafts: DocDraft[] = [];
   175→    for (const file of Array.from(files).slice(0, 5)) {
   176→      if (file.size > 8 * 1024 * 1024) { toast(`"${file.name}" supera gli 8 MB ed è stato scartato.`, "err"); continue; }
   177→      drafts.push({ kind, name: file.name, mime: file.type || "application/octet-stream", data: await readFile(file) });
   178→    }
   179→    set([...current, ...drafts].slice(0, 8));
   180→  };
   181→
   182→  const submit = async () => {
   183→    setErr(null); setBusy(true);
   184→    const res = await api<{ status: string }>("/api/consulente/apply", {
   185→      body: {
   186→        ...f,
   187→        photo,
   188→        zone: zone ? { name: zone.name, province: zone.province, region: zone.region, lat: zone.lat, lng: zone.lng } : undefined,
   189→        docs: [...docsQ, ...docsD, ...docsE],
   190→      },
   191→    });
   192→    setBusy(false);
   193→    if (res.ok) {
   194→      toast("Domanda inviata: lo staff la esaminerà.");
   195→      onDone();
   196→    } else setErr(res.error);
   197→  };
   198→
   199→  return (
   200→    <div className="card">
   201→      <h2 style={{ marginTop: 0 }}>Iscriviti come Consulente Tecnico</h2>
   202→      <p className="small muted">Tutti i dati indicati con * sono obbligatori. La domanda è esaminata dallo staff: la scheda pubblica nasce solo dopo l&apos;approvazione.</p>
   203→      <form onSubmit={e => { e.preventDefault(); void submit(); }}>
   204→        <div className="row">
   205→          <div className="field" style={{ flex: 1, minWidth: 140 }}>
   206→            <label htmlFor="cz-first">Nome *</label>
   207→            <input id="cz-first" required value={f.firstName} onChange={e => setF({ ...f, firstName: e.target.value })} />
   208→          </div>
   209→          <div className="field" style={{ flex: 1, minWidth: 140 }}>
   210→            <label htmlFor="cz-last">Cognome *</label>
   211→            <input id="cz-last" required value={f.lastName} onChange={e => setF({ ...f, lastName: e.target.value })} />
   212→          </div>
   213→        </div>
   214→        <div className="row">
   215→          <div className="field" style={{ flex: 1, minWidth: 160 }}>
   216→            <label htmlFor="cz-prof">Professione *</label>
   217→            <select id="cz-prof" required value={f.profession} onChange={e => setF({ ...f, profession: e.target.value })}>
   218→              <option value="">Scegli…</option>
   219→              {PROFESSIONS.map(p => <option key={p} value={p}>{p}</option>)}
   220→            </select>
   221→          </div>
   222→          <div className="field" style={{ flex: 1, minWidth: 160 }}>
   223→            <label htmlFor="cz-title">Titolo di studio *</label>
   224→            <input id="cz-title" required placeholder="Es. Diploma di geometra / Laurea in ingegneria edile" value={f.title} onChange={e => setF({ ...f, title: e.target.value })} />
   225→          </div>
   226→        </div>
   227→        <PlaceSearch
   228→          id="cz-zone" label="Zona di esercizio *" hint="La località da cui operi: qualsiasi comune d'Italia. La scheda sarà visibile in questa zona."
   229→          selected={zone ? zone.name : null}
   230→          onPick={p => { setZone(p); }}
   231→        />
   232→        {zone ? (
   233→          <p className="small mb2">Zona memorizzata: <b>{zone.name}</b>{zone.province ? ` (${zone.province})` : ""}. <button type="button" className="link" onClick={() => setZone(null)}>Cambia</button></p>
   234→        ) : null}
   235→        <div className="row">
   236→          <div className="field" style={{ flex: 1, minWidth: 130 }}>
   237→            <label htmlFor="cz-radius">Raggio di azione (km) *</label>
   238→            <input id="cz-radius" type="number" min={5} max={300} required value={f.radiusKm} onChange={e => setF({ ...f, radiusKm: Number(e.target.value) })} />
   239→            <span className="hint">Tra 5 e 300 km attorno alla zona di esercizio.</span>
   240→          </div>
   241→          <div className="field" style={{ flex: 1, minWidth: 130 }}>
   242→            <label htmlFor="cz-fee">Tariffa richiesta (€)</label>
   243→            <input id="cz-fee" type="text" inputMode="decimal" placeholder="Es. 300 (vuoto = su richiesta)" value={f.feeEuros} onChange={e => setF({ ...f, feeEuros: e.target.value })} />
   244→            <span className="hint">La tariffa di riferimento per una perizia: potrai confermarla o aggiornarla a ogni richiesta.</span>
   245→          </div>
   246→        </div>
   247→        <div className="row">
   248→          <div className="field" style={{ flex: 1, minWidth: 140 }}>
   249→            <label htmlFor="cz-vat">Partita IVA *</label>
   250→            <input id="cz-vat" required placeholder="Es. 01234567890" value={f.vatNumber} onChange={e => setF({ ...f, vatNumber: e.target.value })} />
   251→          </div>
   252→          <div className="field" style={{ flex: 1, minWidth: 180 }}>
   253→            <label htmlFor="cz-iban">IBAN (accredito compensi) *</label>
   254→            <input id="cz-iban" required placeholder="ITXX..." value={f.iban} onChange={e => setF({ ...f, iban: e.target.value })} />
   255→            <span className="hint">Visibile solo allo staff: serve per accreditarti i compensi.</span>
   256→          </div>
   257→        </div>
   258→        <div className="field">
   259→          <label htmlFor="cz-phone">Telefono (facoltativo)</label>
   260→          <input id="cz-phone" value={f.phone} onChange={e => setF({ ...f, phone: e.target.value })} />
   261→        </div>
   262→        <div className="field">
   263→          <label htmlFor="cz-bio">Presentazione ed esperienza</label>
   264→          <textarea id="cz-bio" rows={4} placeholder="Es. Geometra dal 2008, specializzato in perizie giudiziarie e valutazioni immobiliari nel milanese…" value={f.bio} onChange={e => setF({ ...f, bio: e.target.value })} />
   265→        </div>
   266→        <div className="field">
   267→          <label htmlFor="cz-photo">Foto profilo *</label>
   268→          <input id="cz-photo" type="file" accept="image/*" onChange={async e => {
   269→            const file = e.target.files?.[0];
   270→            if (file) setPhoto(await readFile(file));
   271→          }} />
   272→          <span className="hint">Ricompressa lato server senza metadati (come le foto delle recensioni).</span>
   273→          {photo ? <img src={photo} alt="Anteprima della foto profilo" style={{ width: 96, height: 96, objectFit: "cover", borderRadius: 8, marginTop: 8 }} /> : null}
   274→        </div>
   275→        <fieldset>
   276→          <legend>Documenti (obbligatori, visibili solo allo staff) *</legend>
   277→          <div className="field">
   278→            <label htmlFor="cz-doc-q">Qualifica: titolo, iscrizione all&apos;albo o certificato *</label>
   279→            <input id="cz-doc-q" type="file" multiple onChange={e => void pickFiles(e.target.files, "qualifica", setDocsQ, docsQ)} />
   280→            {docsQ.length ? <p className="small muted mb0">{docsQ.map((d, i) => <span key={i} className="pill pill-gray" style={{ marginRight: 6 }}>{d.name}</span>)}</p> : null}
   281→          </div>
   282→          <div className="field">
   283→            <label htmlFor="cz-doc-d">Documento d&apos;identificazione *</label>
   284→            <input id="cz-doc-d" type="file" multiple onChange={e => void pickFiles(e.target.files, "documento", setDocsD, docsD)} />
   285→            {docsD.length ? <p className="small muted mb0">{docsD.map((d, i) => <span key={i} className="pill pill-gray" style={{ marginRight: 6 }}>{d.name}</span>)}</p> : null}
   286→          </div>
   287→          <div className="field">
   288→            <label htmlFor="cz-doc-e">Esperienza: curriculum, portfolio, referenze (facoltativo)</label>
   289→            <input id="cz-doc-e" type="file" multiple onChange={e => void pickFiles(e.target.files, "esperienza", setDocsE, docsE)} />
   290→            {docsE.length ? <p className="small muted mb0">{docsE.map((d, i) => <span key={i} className="pill pill-gray" style={{ marginRight: 6 }}>{d.name}</span>)}</p> : null}
   291→          </div>
   292→        </fieldset>
   293→        {err ? <p className="error-msg" style={{ display: "block" }}>{err}</p> : null}
   294→        <button className="btn btn-primary" type="submit" disabled={busy}>{busy ? "Invio…" : "Invia la domanda di iscrizione"}</button>
   295→      </form>
   296→    </div>
   297→  );
   298→}

   361→        <button className="btn btn-primary" onClick={() => void saveEdit()} disabled={busy}>{busy ? "Salvo…" : "Salva"}</button>
   362→      </div>

   364→      <h2>Richieste ricevute</h2>
   365→      {incoming.length ? incoming.map(r => <RequestCard key={r.id} r={r} role="consultant" onChanged={onChanged} />) : (

   367→      )}
   368→    </>
   369→  );
   370→}

   376→function RequestCard({ r, role, onChanged, payment }: { r: Richiesta; role: "user" | "consultant"; onChanged: () => void; payment?: { iban: string; instructions: string } }) {
   377→  const { toast } = useApp();
   378→  const [busy, setBusy] = useState(false);
   379→  const [fee, setFee] = useState(r.feeCents != null ? String(r.feeCents / 100) : "");
   380→  const [rejectReason, setRejectReason] = useState("");
   381→  const [title, setTitle] = useState(r.reportTitle || "");
   382→  const [body, setBody] = useState("");
   383→  const [open, setOpen] = useState<string | null>(null);
   384→
   385→  const act = async (action: string, extra: Record<string, unknown> = {}) => {
   386→    setBusy(true);
   387→    const res = await api(`/api/consulente/requests/${r.id}/action`, { body: { action, ...extra } });
   388→    setBusy(false);
   389→    if (res.ok) { toast("Operazione registrata."); onChanged(); setOpen(null); }
   390→    else toast(res.error, "err");
   391→  };
   392→
   393→  return (
   394→    <article className="card mb2">
   395→      <div className="row" style={{ justifyContent: "space-between", alignItems: "baseline", flexWrap: "wrap", gap: 8 }}>
   396→        <h3 style={{ margin: 0 }}>{r.kind} · {r.address}</h3>
   397→        <StatusPill status={r.status} />
   398→      </div>
   399→      <p className="small muted" style={{ marginTop: 6 }}>
   400→        {role === "consultant" ? <>Da: {r.counterpart.nickname}</> : <>Consulente: {r.counterpart.name} ({r.counterpart.profession})</>}
   401→        {" · "}{fmtDateTime(r.createdAt)}
   402→        {r.feeCents != null ? <> · tariffa € {fmtEuro(r.feeCents)}</> : null}
   403→        {r.status === "delivered" || r.status === "closed" ? <> · tuo accredito € {fmtEuro(r.payoutCents)} (commissione € {fmtEuro(r.commissionCents)})</> : null}
   404→      </p>
   405→      <p className="small" style={{ whiteSpace: "pre-wrap" }}>{r.description}</p>
   406→      {r.rejectReason ? <p className="small muted">Motivo del rifiuto: {r.rejectReason}</p> : null}
   407→
   408→      {role === "consultant" && r.canAccept ? (
   409→        <div className="card card-pad-s mb1">
   410→          <div className="row" style={{ alignItems: "flex-end" }}>
   411→            <div className="field mb0" style={{ flex: 1, minWidth: 120 }}>
   412→              <label htmlFor={`fee-${r.id}`}>Tariffa confermata (€) *</label>
   413→              <input id={`fee-${r.id}`} value={fee} inputMode="decimal" onChange={e => setFee(e.target.value)} />
   414→            </div>
   415→            <button className="btn btn-primary" disabled={busy} onClick={() => void act("accept", { feeEuros: fee })}>Accetti l&apos;incarico</button>
   416→          </div>
   417→          {open === "reject" ? (
   418→            <div className="field mt1">
   419→              <label htmlFor={`rej-${r.id}`}>Motivo del rifiuto *</label>
   420→              <textarea id={`rej-${r.id}`} rows={2} value={rejectReason} onChange={e => setRejectReason(e.target.value)} />
   421→              <div className="row mt1"><button className="btn btn-danger" disabled={busy || rejectReason.trim().length < 5} onClick={() => void act("reject", { reason: rejectReason })}>Conferma rifiuto</button><button className="btn btn-ghost" onClick={() => setOpen(null)}>Annulla</button></div>
   422→            </div>
   423→          ) : (
   424→            <button className="link small" onClick={() => setOpen("reject")} style={{ marginTop: 6 }}>oppure rifiuta con motivazione…</button>
   425→          )}
   426→        </div>
   427→      ) : null}
   428→
   429→      {role === "consultant" && r.canDeliver ? (
   430→        <div className="card card-pad-s mb1">
   431→          <p className="small mb1"><b>Consegna la relazione.</b> Verrà resa come PDF scaricabile dal richiedente; l&apos;accredito ti verrà riconosciuto alla consegna.</p>
   432→          <div className="field">
   433→            <label htmlFor={`rt-${r.id}`}>Titolo della relazione *</label>
   434→            <input id={`rt-${r.id}`} value={title} onChange={e => setTitle(e.target.value)} placeholder="Es. Relazione di valutazione immobiliare" />
   435→          </div>
   436→          <div className="field">
   437→            <label htmlFor={`rb-${r.id}`}>Testo della relazione * (almeno 300 caratteri)</label>
   438→            <textarea id={`rb-${r.id}`} rows={8} value={body} onChange={e => setBody(e.target.value)} placeholder={"Struttura suggerita:\n1. Premessa e incarico\n2. Descrizione dell'immobile\n3. Verifiche eseguite\n4. Valutazione / esito\n5. Conclusioni"} />
   439→          </div>
   440→          <button className="btn btn-primary" disabled={busy || title.trim().length < 5 || body.trim().length < 300} onClick={() => void act("deliver", { title, body })}>Consegna la relazione</button>
   441→        </div>
   442→      ) : null}
   443→
   444→      {role === "user" && r.status === "accepted" ? (
   445→        <div className="card card-pad-s mb1">
   446→          <p className="small mb1"><b>Pagamento della tariffa (€ {fmtEuro(r.feeCents)}).</b> Il consulente procede alla relazione dopo la registrazione del pagamento da parte della piattaforma.</p>
   447→          {payment?.iban ? <p className="small">IBAN della piattaforma: <b className="mono">{payment.iban}</b></p> : null}
   448→          {payment?.instructions ? <p className="small" style={{ whiteSpace: "pre-wrap" }}>{payment.instructions}</p> : (
   449→            <p className="small muted">Le istruzioni di pagamento (IBAN e causale) sono indicate qui dal gestore della piattaforma; in caso contrario contatta l&apos;amministrazione dal modulo contatti.</p>
   450→          )}
   451→          <p className="small muted mb0">Causale consigliata: <span className="mono">perizia {r.id}</span></p>
   452→        </div>
   453→      ) : null}
   454→
   455→      {r.status === "delivered" || r.status === "closed" ? (
   456→        r.reportTitle ? <p className="small mb1"><b>Relazione consegnata:</b> {r.reportTitle}</p> : null
   457→      ) : null}
   458→
   459→      <div className="row">
   460→        {r.status === "delivered" || r.status === "closed" ? (
   461→          <a className="btn btn-primary btn-sm" href={`/api/consulente/requests/${r.id}/pdf`}>Scarica il PDF</a>
   462→        ) : null}
   463→        {role === "user" && r.canCancel ? (
   464→          <button className="btn btn-ghost btn-sm" disabled={busy} onClick={() => void act("cancel")}>Annulla la richiesta</button>
   465→        ) : null}
   466→      </div>
   467→    </article>
   468→  );
   469→}
