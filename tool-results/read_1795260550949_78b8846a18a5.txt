    16→  const salt = randomBytes(16).toString("hex");
    17→  const hash = scryptSync(password, salt, 64, { N: 16384, r: 8, p: 1 }).toString("hex");

    23→function mulberry32(seed: number) {
    24→  let a = seed >>> 0;
    25→  return function () {
    26→    a |= 0; a = (a + 0x6D2B79F5) | 0;
    27→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    28→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    29→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    30→  };
    31→}

   513→import { createCipheriv } from "crypto";
   514→function encryptSecretStatic(plain: string): string {
   515→  const APP_SECRET = process.env.APP_SECRET || "";
   516→  if (!APP_SECRET) throw new Error("APP_SECRET mancante in .env");
   517→  const key = createHash("sha256").update(APP_SECRET).digest();
   518→  const iv = randomBytes(12);
   519→  const cipher = createCipheriv("aes-256-gcm", key, iv);
   520→  const enc = Buffer.concat([cipher.update(plain, "utf8"), cipher.final()]);
   521→  return `${iv.toString("hex")}.${cipher.getAuthTag().toString("hex")}.${enc.toString("hex")}`;
   522→}
   523→function randomBytesStatic(r: () => number, n: number): Buffer {
   524→  // per il seed serve entropia reale per i segreti TOTP, non RNG deterministico
   525→  return randomBytes(n);
   526→}
   527→function base32(bytes: Buffer): string {
   528→  const B32 = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
   529→  let bits = 0, val = 0, out = "";
   530→  for (const b of bytes) {
   531→    val = (val << 8) | b; bits += 8;
   532→    while (bits >= 5) { out += B32[(val >>> (bits - 5)) & 31]; bits -= 5; }
   533→  }
   534→  if (bits > 0) out += B32[(val << (5 - bits)) & 31];
   535→  return out;
   536→}
