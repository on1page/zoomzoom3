    18→  const salt = randomBytes(16).toString("hex");
    19→  const hash = scryptSync(password, salt, 64, { N: 16384, r: 8, p: 1 }).toString("hex");

    25→function mulberry32(seed: number) {
    26→  let a = seed >>> 0;
    27→  return function () {
    28→    a |= 0; a = (a + 0x6D2B79F5) | 0;
    29→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    30→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    31→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    32→  };

   629→function encryptSecretStatic(plain: string): string {
   630→  const APP_SECRET = process.env.APP_SECRET || "";
   631→  if (!APP_SECRET) throw new Error("APP_SECRET mancante in .env");
   632→  const key = createHash("sha256").update(APP_SECRET).digest();
   633→  const iv = randomBytes(12);
   634→  const cipher = createCipheriv("aes-256-gcm", key, iv);
   635→  const enc = Buffer.concat([cipher.update(plain, "utf8"), cipher.final()]);
   636→  return `${iv.toString("hex")}.${cipher.getAuthTag().toString("hex")}.${enc.toString("hex")}`;
   637→}

   642→function base32(bytes: Buffer): string {
   643→  const B32 = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
   644→  let bits = 0, val = 0, out = "";
   645→  for (const b of bytes) {
   646→    val = (val << 8) | b; bits += 8;
   647→    while (bits >= 5) { out += B32[(val >>> (bits - 5)) & 31]; bits -= 5; }
   648→  }
   649→  if (bits > 0) out += B32[(val << (5 - bits)) & 31];
   650→  return out;
   651→}
