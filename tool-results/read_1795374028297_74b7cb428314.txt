   184→/* ---------- Stepper: 1 Soggetto · 2 Luogo · 3 Scheda · 4 Recensione ---------- */
   185→function Stepper({ labels, current, onJump }: { labels: string[]; current: number; onJump: (i: number) => void }) {
   186→  return (
   187→    <ol className="zz-stepper" aria-label="Avanzamento">
   188→      {labels.map((l, i) => (
   189→        <Fragment key={l}>
   190→          {i > 0 ? <span className="zz-step-arrow" aria-hidden="true">›</span> : null}
   191→          <li
   192→            className={`zz-step${i === current ? " on" : i < current ? " done" : ""}`}
   193→            aria-current={i === current ? "step" : undefined}
   194→            onClick={i < current ? () => onJump(i) : undefined}
   195→            onKeyDown={i < current ? (e => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); onJump(i); } }) : undefined}
   196→            tabIndex={i < current ? 0 : undefined}
   197→            title={i < current ? `Torna a: ${l}` : undefined}
   198→          >
   199→            <span className="zz-dot">{i < current ? <CheckSvg size={11} /> : i + 1}</span>
   200→            {l}
   201→          </li>
   202→        </Fragment>
   203→      ))}
   204→    </ol>
   205→  );
   206→}

   275→/* ---------- Passo 2: il luogo, un solo campo ---------- */
   276→function PlaceStep({ target, onBack, onInternal, onDraft }: {
   277→  target: Target;
   278→  onBack: () => void;
   279→  onInternal: (ent: SelEntity, note: { created: boolean; reused: boolean }) => void;
   280→  onDraft: (d: Draft) => void;
   281→}) {
   282→  const { boot, toast } = useApp();
   283→  const [q, setQ] = useState("");
   284→  const [internal, setInternal] = useState<SearchRow[] | null>(null);
   285→  const [geoRes, setGeoRes] = useState<GeoRow[] | null>(null);
   286→  const [busyInt, setBusyInt] = useState(false);
   287→  const [busyGeo, setBusyGeo] = useState(false);
   288→  const [throttled, setThrottled] = useState(false);
   289→  const [gpsStage, setGpsStage] = useState<"idle" | "gps" | "addr">("idle");
   290→  const [zoneBusy, setZoneBusy] = useState(false);
   291→  const [pubZones, setPubZones] = useState<PubZone[] | null>(null);
   292→  const [personBlock, setPersonBlock] = useState<SearchRow | null>(null);
   293→
   294→  const zoneName = useCallback((id: string | null) => boot?.zones.find(z => z.id === id)?.name || "", [boot]);
   295→
   296→  // zona: elenco delle città già sul catalogo (con la loro scheda, se attiva)
   297→  useEffect(() => {
   298→    if (target !== "zona") return;
   299→    void (async () => {
   300→      const res = await api<{ zones: PubZone[] }>("/api/public/map");
   301→      setPubZones(res.ok ? res.zones : []);
   302→    })();
   303→  }, [target]);
   304→
   305→  // ricerca combinata: schede interne + luoghi della mappa
   306→  const runSearch = useCallback(async (query: string) => {
   307→    if (target !== "zona") {
   308→      setBusyInt(true);
   309→      const cat = target === "immobile" ? "immobile" : SUBJECT_CATS.join(",");
   310→      const res = await api<{ rows: SearchRow[] }>(`/api/public/search?cat=${cat}${query ? `&q=${encodeURIComponent(query)}` : ""}`);
   311→      setInternal(res.ok ? res.rows : []);
   312→      setBusyInt(false);
   313→    }
   314→    if (target !== "altro" && query.length >= 3) {
   315→      setBusyGeo(true);
   316→      const res = await api<{ results: GeoRow[]; throttled?: boolean }>(`/api/public/geocode?q=${encodeURIComponent(query)}`);
   317→      setThrottled(!!(res as { throttled?: boolean }).throttled);
   318→      setGeoRes(res.ok ? res.results : []);
   319→      setBusyGeo(false);
   320→    } else {
   321→      setGeoRes(null);
   322→    }
   323→  }, [target]);
   324→
   325→  // ricerca con debounce. A vuoto (immobile) si aprono le schede esistenti:
   326→  // chi entra trova subito anche il catalogo, non solo la mappa (C5)
   327→  useEffect(() => {
   328→    const query = q.trim();
   329→    if (query.length < 2) {
   330→      setGeoRes(null); setThrottled(false);
   331→      if (target === "immobile") void runSearch(""); else setInternal(null);
   332→      return;
   333→    }
   334→    const t = setTimeout(() => { void runSearch(query); }, 420);
   335→    return () => clearTimeout(t);
   336→  }, [q, target, runSearch]);
   337→
   338→  // scheda interna già presente: la si usa così com'è (USE del diagramma)
   339→  const pickInternal = (r: SearchRow) => {
   340→    const cat = boot?.categories.find(c => c.id === r.entity.categoryId);
   341→    if (cat?.isPerson) { setPersonBlock(r); return; }
   342→    onInternal({ id: r.entity.id, name: r.entity.name, categoryId: r.entity.categoryId }, { created: false, reused: true });
   343→  };
   344→
   345→  // città già sul catalogo: scheda attiva → si usa; altrimenti si attiva ora
   346→  const pickZone = async (z: { id?: string; name: string; lat: number; lng: number; entityId?: string | null }) => {
   347→    if (z.entityId) { onInternal({ id: z.entityId, name: z.name, categoryId: "zona" }, { created: false, reused: true }); return; }
   348→    setZoneBusy(true);
   349→    const res = await api<{ entityId: string; name: string; created: boolean }>("/api/zones", {
   350→      body: { name: z.name, province: null, region: null, lat: z.lat, lng: z.lng },
   351→    });
   352→    setZoneBusy(false);
   353→    if (res.ok && res.entityId) {
   354→      toast(res.created ? `Scheda di ${z.name} attivata.` : `Scheda di ${z.name} già pronta.`);
   355→      onInternal({ id: res.entityId, name: z.name, categoryId: "zona" }, { created: res.created, reused: !res.created });
   356→    } else {
   357→      toast("Non riesco ad attivare la scheda della città: riprova tra poco.", "err");
   358→    }
   359→  };
   360→
   361→  // luogo della mappa: indirizzo o città → minibozza che finisce al passo 3.
   362→  // Via e civico restano SEPARATI: in un civico possono esserci più unità
   363→  // (condomini) e la via è la base del confronto con le schede esistenti
   364→  const pickGeo = (g: GeoRow) => {
   365→    onDraft({
   366→      address: g.road || "",
   367→      civico: g.houseNumber || "",
   368→      unit: "",
   369→      zone: g.zoneName ? { name: g.zoneName, province: g.province, region: g.region, lat: g.lat, lng: g.lng } : null,
   370→      geo: { lat: g.lat, lng: g.lng },
   371→    });
   372→  };
   373→
   374→  // GPS + reverse geocoding: chi è sul posto preme un bottone e il passo 3
   375→  // si riempie da solo (indirizzo reale, città, pin). Coordinate al geocoder
   376→  // pubblico SOLO su azione esplicita.
   377→  const useGps = () => {
   378→    if (!navigator.geolocation) { toast("Questo browser non espone la posizione: usa la ricerca.", "err"); return; }
   379→    setGpsStage("gps");
   380→    navigator.geolocation.getCurrentPosition(
   381→      async pos => {
   382→        const pt = { lat: Math.round(pos.coords.latitude * 1e6) / 1e6, lng: Math.round(pos.coords.longitude * 1e6) / 1e6 };
   383→        setGpsStage("addr");
   384→        const res = await api<{ found: boolean; address: { road: string | null; houseNumber: string | null; zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${pt.lat}&lng=${pt.lng}`);
   385→        setGpsStage("idle");
   386→        if (res.ok && res.found && res.address) {
   387→          const a = res.address;
   388→          onDraft({
   389→            address: a.road || "",
   390→            civico: a.houseNumber || "",
   391→            unit: "",
   392→            zone: a.zoneName ? { name: a.zoneName, province: a.province, region: a.region, lat: pt.lat, lng: pt.lng } : null,
   393→            geo: pt,
   394→          });
   395→        } else {
   396→          onDraft({ address: "", civico: "", unit: "", zone: null, geo: pt });

   650→/* ---------- Passo 3: controlla la scheda (badge + solo i mancanti) ---------- */
   651→function ConfirmStep({ target, draft, setDraft, onBack, onReady }: {
   652→  target: Target;
   653→  draft: Draft;
   654→  setDraft: (updater: (d: Draft) => Draft) => void;
   655→  onBack: () => void;
   656→  onReady: (ent: SelEntity, note: { created: boolean; reused: boolean }) => void;
   657→}) {
   658→  const { boot, toast } = useApp();
   659→  const [editZone, setEditZone] = useState(!draft.zone);
   660→  const [editAddr, setEditAddr] = useState(target === "immobile" && draft.address.trim().length < 3);
   661→  const [editCivico, setEditCivico] = useState(target === "immobile" && !draft.civico.trim());
   662→  const [editUnit, setEditUnit] = useState(false);
   663→  const [showMap, setShowMap] = useState(false);
   664→  const [busy, setBusy] = useState(false);
   665→  const [err, setErr] = useState<string | null>(null);
   666→  const [nearby, setNearby] = useState<NearbyRow[] | null>(null);
   667→  // schede già esistenti allo stesso indirizzo (via + civico): il condominio
   668→  // si vede PRIMA di crearne una nuova — ogni unità ha la sua scheda
   669→  const [building, setBuilding] = useState<BuildingRow[] | null>(null);
   670→
   671→  const isImmobile = target === "immobile";
   672→  const via = draft.address.trim();
   673→  const civico = draft.civico.trim();
   674→  const unit = draft.unit.trim();
   675→
   676→  // la città scelta dalla mappa può già esistere nel catalogo: se c'è, il suo
   677→  // id locale serve al controllo delle schede dello stesso civico (building)
   678→  useEffect(() => {
   679→    const name = draft.zone?.name?.toLowerCase();
   680→    if (!name || draft.zone?.localId || !boot?.zones) return;
   681→    const known = boot.zones.find(z => z.name.toLowerCase() === name);
   682→    if (known) setDraft(d => (d.zone && !d.zone.localId ? { ...d, zone: { ...d.zone, localId: known.id } } : d));
   683→  }, [draft.zone, boot, setDraft]);
   684→  const addrOk = via.length >= 3;
   685→  const civicoOk = civico.length >= 1;
   686→  const valid = !!draft.zone && (!isImmobile || (addrOk && civicoOk));
   687→  const myUnitNorm = normUnit(unit);
   688→  const fullAddr = dispAddress(via, civico);
   689→
   690→  // conferma: proposta di riuso (C6, pin) → schede dello stesso civico →
   691→  // creazione/dedup (EXIST/CREATE)
   692→  const confirm = async (skip: { prox?: boolean; building?: boolean } = {}) => {
   693→    if (busy || !valid) return;
   694→    setErr(null); setBusy(true);
   695→
   696→    if (target === "zona") {
   697→      const z = draft.zone!;
   698→      const res = await api<{ entityId: string; created: boolean }>("/api/zones", {
   699→        body: { name: z.name, province: z.province, region: z.region, lat: z.lat, lng: z.lng },
   700→      });
   701→      setBusy(false);
   702→      if (res.ok && res.entityId) {
   703→        toast(res.created ? `Scheda di ${z.name} attivata.` : `Scheda di ${z.name} già pronta.`);
   704→        onReady({ id: res.entityId, name: z.name, categoryId: "zona" }, { created: res.created, reused: !res.created });
   705→      } else setErr(res.error);
   706→      return;
   707→    }
   708→
   709→    // immobile — con il pin, prima propongo le schede vicine (entro 50 m)
   710→    if (draft.geo && !skip.prox) {
   711→      const nr = await api<{ rows: NearbyRow[] }>(`/api/public/nearby?lat=${draft.geo.lat}&lng=${draft.geo.lng}${draft.zone?.localId ? `&zona=${draft.zone.localId}` : ""}`);
   712→      if (nr.ok && nr.rows?.length) { setNearby(nr.rows); setBusy(false); return; }
   713→    }
   714→    setNearby(null);
   715→
   716→    // stesso indirizzo (via + civico): le schede del condominio si vedono
   717→    // prima di crearne una nuova. Il civico discrimina gli edifici nella
   718→    // via, l'interno le unità nello stesso civico: se una coincide, si usa
   719→    if (!skip.building && draft.zone?.localId && addrOk) {
   720→      const br = await api<{ rows: BuildingRow[] }>(`/api/public/building?zona=${encodeURIComponent(draft.zone.localId)}&via=${encodeURIComponent(via)}&civico=${encodeURIComponent(civico)}`);
   721→      if (br.ok && br.rows?.length) { setBuilding(br.rows); setBusy(false); return; }
   722→    }
   723→    setBuilding(null);
   724→
   725→    const z = draft.zone;
   726→    const res = await api<{ entity: { id: string; name: string } }>("/api/entities", {
   727→      body: {
   728→        address: via,
   729→        streetNumber: civico,
   730→        unit: unit || undefined,
   731→        zoneId: z?.localId,
   732→        zone: z && !z.localId
   733→          ? { name: z.name, province: z.province, region: z.region, lat: z.lat, lng: z.lng }
   734→          : undefined,
   735→        geo: draft.geo ?? undefined,
   736→      },
   737→    });
   738→    setBusy(false);
   739→    if (res.ok && res.entity) {
   740→      onReady({ id: res.entity.id, name: res.entity.name, categoryId: "immobile" }, { created: true, reused: false });
   741→      return;
   742→    }
   743→    const ex = (res as { existingId?: string }).existingId;
   744→    if (ex) {
   745→      toast("Immobile già presente: uso la scheda esistente.");
   746→      onReady({ id: ex, name: unit ? `${fullAddr}, ${unit}` : fullAddr, categoryId: "immobile" }, { created: false, reused: true });
   747→      return;
   748→    }
   749→    setErr(res.error);
   750→  };
   751→
   752→  return (
   753→    <div className="container" style={{ maxWidth: 760, paddingTop: 18 }}>
   754→      <div className="section-head">
   755→        <div>
   756→          <h1 style={{ margin: 0 }}>Controlla la scheda</h1>
   757→          <p className="sub" style={{ margin: "6px 0 0" }}>
   758→            {isImmobile
   759→              ? <>Serve <span className="pill pill-green">città</span>, <span className="pill pill-green">via</span> e <span className="pill pill-green">civico</span>: il resto è facoltativo.</>
   760→              : <>Serve la <span className="pill pill-green">località</span>: il resto arriva con la recensione.</>}
   761→          </p>
   762→        </div>
   763→        <button className="btn btn-ghost btn-sm" onClick={onBack}>← Cambia luogo</button>
   764→      </div>
   765→
   766→      <div className="card">
   767→        <div className="zz-facts">
   768→          {/* --- città / località --- */}
   769→          <div className="zz-fact">
   770→            <span className="zz-fact-k">{target === "zona" ? "Località" : "Città"}</span>
   771→            {draft.zone && !editZone ? (
   772→              <>
   773→                <span className="zz-fact-v ok"><CheckSvg size={11} /> {draft.zone.name}{draft.zone.province ? ` (${draft.zone.province})` : ""}</span>
   774→                <span className="spacer" />
   775→                <button type="button" className="btn btn-ghost btn-xs" onClick={() => setEditZone(true)}>Modifica</button>
   776→              </>
   777→            ) : (
   778→              <div style={{ flex: 1, minWidth: 220 }}>
   779→                <PlaceSearch
   780→                  id="cf-city" localZones={boot?.zones} placeholder="Città, comune o paese…"
   781→                  selected={null}
   782→                  onPick={p => {
   783→                    setDraft(d => ({ ...d, zone: { name: p.zoneName || p.name, province: p.province, region: p.region, lat: p.lat, lng: p.lng, localId: p.localId } }));
   784→                    setEditZone(false);
   785→                  }}
   786→                />
   787→              </div>
   788→            )}
   789→          </div>
   790→
   791→          {/* --- via (solo immobile): il civico va nel campo suo --- */}

   975→    </div>
   976→  );
