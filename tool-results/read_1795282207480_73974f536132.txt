   154→type ZonePick = {

   165→function FindStep({ target, onSelected, onBack }: {
   166→  target: "immobile" | "zona";
   167→  onSelected: (ent: SelEntity) => void;
   168→  onBack: () => void;
   169→}) {
   170→  const { boot, toast } = useApp();
   171→  const [q, setQ] = useState("");
   172→  const [rows, setRows] = useState<SearchRow[] | null>(null);
   173→  const [busy, setBusy] = useState(false);
   174→
   175→  // stato "aggiungi immobile"
   176→  const [adding, setAdding] = useState(false);
   177→  const [address, setAddress] = useState("");
   178→  const [zone, setZone] = useState<ZonePick | null>(null);
   179→  const [geo, setGeo] = useState<LatLng | null>(null);
   180→  const [geoQ, setGeoQ] = useState("");
   181→  const [geoRes, setGeoRes] = useState<Array<{ label: string; lat: number; lng: number; zoneName: string | null; province: string | null; region: string | null; road?: string | null; houseNumber?: string | null }> | null>(null);
   182→  const [geoBusy, setGeoBusy] = useState(false);
   183→  const [gpsStage, setGpsStage] = useState<"idle" | "gps" | "addr">("idle");
   184→  const [addErr, setAddErr] = useState<string | null>(null);
   185→
   186→  // zona: elenco delle città già sul catalogo (dalla mappa pubblica)
   187→  const [pubZones, setPubZones] = useState<PubZone[] | null>(null);
   188→  const [zoneBusy, setZoneBusy] = useState(false);
   189→
   190→  const search = useCallback(async (query: string) => {
   191→    setBusy(true);
   192→    const res = await api<{ rows: SearchRow[] }>(`/api/public/search?cat=immobile${query ? `&q=${encodeURIComponent(query)}` : ""}`);
   193→    setRows(res.ok ? res.rows : []);
   194→    setBusy(false);
   195→  }, []);
   196→
   197→  useEffect(() => {
   198→    if (target === "immobile") void search("");
   199→    else void (async () => {
   200→      const res = await api<{ zones: PubZone[] }>("/api/public/map");
   201→      setPubZones(res.ok ? res.zones : []);
   202→    })();
   203→  }, [target, search]);
   204→
   205→  // attiva la scheda recensibile di una città: se esiste già si usa quella,
   206→  // altrimenti nasce adesso (copertura nazionale, paesi inclusi)
   207→  const activateZone = async (z: { id?: string; name: string; lat: number; lng: number; province?: string | null; region?: string | null; entityId?: string | null }) => {
   208→    if (z.entityId) { onSelected({ id: z.entityId, name: z.name, categoryId: "zona" }); return; }
   209→    setZoneBusy(true);
   210→    const res = await api<{ entityId: string; name: string; created: boolean }>("/api/zones", {
   211→      body: { name: z.name, province: z.province ?? null, region: z.region ?? null, lat: z.lat, lng: z.lng },
   212→    });
   213→    setZoneBusy(false);
   214→    if (res.ok && res.entityId) {
   215→      toast(res.created ? `Scheda di ${z.name} attivata: ora scrivi la recensione.` : `Scheda di ${z.name} già pronta: ora scrivi la recensione.`);
   216→      onSelected({ id: res.entityId, name: z.name, categoryId: "zona" });
   217→    } else {
   218→      toast("Non riesco ad attivare la scheda della città: riprova tra poco.", "err");
   219→    }
   220→  };
   221→
   222→  const doGeocode = async () => {
   223→    if (geoQ.trim().length < 3) return;
   224→    setGeoBusy(true); setGeoRes(null);
   225→    const res = await api<{ results: Array<{ label: string; lat: number; lng: number; zoneName: string | null; province: string | null; region: string | null; road?: string | null; houseNumber?: string | null }> }>(`/api/public/geocode?q=${encodeURIComponent(geoQ.trim())}`);
   226→    setGeoRes(res.ok ? res.results : []);
   227→    setGeoBusy(false);
   228→  };
   229→
   230→  // GPS + reverse geocoding: chi è davanti all'immobile preme un bottone e
   231→  // il modulo si riempie da solo (indirizzo reale, città, pin). Le coordinate
   232→  // vanno al geocoder pubblico SOLO in questo momento, su richiesta esplicita.
   233→  const useGpsAddress = () => {
   234→    if (!navigator.geolocation) { toast("Questo browser non espone la posizione: usa la ricerca o la mappa.", "err"); return; }
   235→    setGpsStage("gps");
   236→    navigator.geolocation.getCurrentPosition(
   237→      async pos => {
   238→        const pt = { lat: Math.round(pos.coords.latitude * 1e6) / 1e6, lng: Math.round(pos.coords.longitude * 1e6) / 1e6 };
   239→        setGeo(pt);
   240→        setGpsStage("addr");
   241→        const res = await api<{ found: boolean; address: { road: string | null; houseNumber: string | null; zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${pt.lat}&lng=${pt.lng}`);
   242→        setGpsStage("idle");
   243→        if (res.ok && res.found && res.address) {
   244→          const a = res.address;
   245→          const street = [a.road, a.houseNumber].filter(Boolean).join(" ");
   246→          if (street) setAddress(street);
   247→          if (a.zoneName) setZone({ name: a.zoneName, province: a.province, region: a.region, lat: pt.lat, lng: pt.lng });
   248→          toast("Indirizzo riconosciuto: controlla e correggi se serve.");
   249→        } else {
   250→          toast("Posizione rilevata, ma l'indirizzo non risulta mappato: inseriscilo a mano (il pin è già a posto).", "err");
   251→        }
   252→      },
   253→      () => { setGpsStage("idle"); toast("Posizione non disponibile: consenti l'accesso dal browser oppure cerca a mano.", "err"); },
   254→      { timeout: 9000, maximumAge: 30000, enableHighAccuracy: true },
   255→    );
   256→  };
   257→
   258→  // conferma di un INDIRIZZO dai suggerimenti del campo indirizzo: riempie
   259→  // via/civico, città e pin in un colpo solo — l'utente vede esattamente
   260→  // cosa sta per salvare e lo conferma scegliendo il suggerimento
   261→  const pickAddress = (p: AddressPlace) => {
   262→    setGeo({ lat: p.lat, lng: p.lng });
   263→    if (p.zoneName) {
   264→      setZone({ name: p.zoneName, province: p.province, region: p.region, lat: p.lat, lng: p.lng });
   265→    }
   266→  };
   267→
   268→  // conferma di un risultato della ricerca sulla mappa: piazza il pin e
   269→  // ricava la città. Se il risultato non porta il comune (es. una via di
   270→  // campagna mappata senza riferimento), lo chiede al reverse geocoder sul
   271→  // punto scelto: un'azione esplicita dell'utente, come il GPS. Prima di
   272→  // questa correzione il pin veniva piazzato ma la città restava vuota,
   273→  // senza spiegazioni: la località trovata non era confermabile.
   274→  const pickGeoResult = async (g: NonNullable<typeof geoRes>[number]) => {
   275→    setGeo({ lat: g.lat, lng: g.lng });
   276→    setGeoRes(null);
   277→    const street = [g.road, g.houseNumber].filter(Boolean).join(" ");
   278→    if (street && !address.trim()) setAddress(street);
   279→    if (zone) return; // città già scelta: il risultato serve solo per il pin
   280→    if (g.zoneName) {
   281→      setZone({ name: g.zoneName, province: g.province, region: g.region, lat: g.lat, lng: g.lng });
   282→      return;
   283→    }
   284→    const res = await api<{ found: boolean; address: { zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${g.lat}&lng=${g.lng}`);
   285→    if (res.ok && res.found && res.address?.zoneName) {
   286→      setZone({ name: res.address.zoneName, province: res.address.province, region: res.address.region, lat: g.lat, lng: g.lng });
   287→    } else {
   288→      toast("Posizione segnata sulla mappa: scegli anche la città nel campo “Città, comune o paese”.", "err");
   289→    }
   290→  };
   291→
   292→  const addEntity = async () => {
   293→    setAddErr(null);
   294→    const res = await api<{ entity: { id: string; name: string } }>("/api/entities", {
   295→      body: {
   296→        address: address.trim(),
   297→        // città già presente → id diretto; nuova località → nome+coordinate.
   298→        // Le coordinate vanno solo se davvero valide (numeri finiti): il
   299→        // server ha comunque i fallback (zona nota col nome, pin sulla mappa)
   300→        zoneId: zone?.localId,
   301→        zone: zone && !zone.localId
   302→          ? {
   303→            name: zone.name, province: zone.province, region: zone.region,
   304→            lat: Number.isFinite(zone.lat) ? zone.lat : undefined,
   305→            lng: Number.isFinite(zone.lng) ? zone.lng : undefined,
   306→          }
   307→          : undefined,
   308→        geo,
   309→      },
   310→    });
   311→    if (res.ok) {
   312→      if (res.entity) {
   313→        toast("Immobile aggiunto: ora scrivi la recensione.");
   314→        onSelected({ id: res.entity.id, name: res.entity.name, categoryId: "immobile" });
   315→      }
   316→      return;
   317→    }
   318→    const ex = (res as { existingId?: string }).existingId;
   319→    if (ex) {
   320→      toast("Questo immobile è già presente: puoi recensire la scheda esistente.");
   321→      onSelected({ id: ex, name: address.trim(), categoryId: "immobile" });
   322→      return;
   323→    }
   324→    setAddErr(res.error);
   325→  };
   326→
   327→  const zoneName = (id: string | null) => boot?.zones.find(z => z.id === id)?.name || "";
   328→
   329→  return (
   330→    <div className="container" style={{ maxWidth: 860, paddingTop: 26 }}>
   331→      <div className="section-head">
   332→        <div>
   333→          <h1 style={{ margin: 0 }}>{target === "immobile" ? "Quale immobile?" : "Quale città?"}</h1>
   334→          <p className="sub">{target === "immobile" ? "Cercalo tra le schede esistenti oppure aggiungilo dall'indirizzo: basta quello." : "Una città già attiva, oppure qualsiasi comune o paese d'Italia: la scheda si attiva al momento."}</p>
   335→        </div>
   336→        <button className="btn btn-ghost btn-sm" onClick={onBack}>← Cambia categoria</button>
   337→      </div>
   338→
   339→      {target === "zona" ? (
   340→        <>
   341→          {pubZones == null ? (
   342→            <p className="muted">Carico le città…</p>
   343→          ) : (
   344→            <div className="grid grid-3">
   345→              {pubZones.map(z => (
   346→                <button key={z.id} className="card card-click" style={{ textAlign: "left" }} disabled={zoneBusy}
   347→                  onClick={() => void activateZone({ id: z.id, name: z.name, lat: z.lat, lng: z.lng, entityId: z.entityId })}>
   348→                  <h3 style={{ margin: 0 }}>{z.name}</h3>
   349→                  <span className="small muted">{z.entityId ? "Scheda attiva · servizi, trasporti, sicurezza, costi" : "Sul catalogo · scheda da attivare"}</span>
   350→                </button>
   351→              ))}
   352→            </div>
   353→          )}
   354→          <div className="card mt3">
   355→            <h2 style={{ marginBottom: 4 }}>La tua città non è tra queste?</h2>
   356→            <p className="small muted" style={{ marginTop: 0 }}>Cerca qualsiasi comune, paese o frazione d&apos;Italia: la scheda nasce ora e resta nel catalogo.</p>
   357→            <PlaceSearch id="zps-city" localZones={boot?.zones}
   358→              placeholder="Es. Corte de' Frati, Marciaga, Alassio…"
   359→              onPick={p => void activateZone({
   360→                name: p.zoneName || p.name,
   361→                province: p.province, region: p.region,
   362→                lat: p.lat, lng: p.lng,
   363→                entityId: p.localId ? (pubZones?.find(z => z.id === p.localId)?.entityId ?? null) : null,
   364→              })} />
   365→            {zoneBusy ? <p className="small muted mb0">Attivo la scheda…</p> : null}
   366→          </div>
   367→        </>
   368→      ) : (
   369→        <>
   370→          <form className="searchbar" role="search" onSubmit={e => { e.preventDefault(); void search(q.trim()); }}>
   371→            <input type="search" placeholder="Cerca per indirizzo o nome (es. Via del Filatoio)" aria-label="Cerca immobile"
   372→              value={q} onChange={e => setQ(e.target.value)} />
   373→            <button className="btn btn-primary" type="submit" disabled={busy}>{busy ? "Cerca…" : "Cerca"}</button>
   374→            <button type="button" className="btn btn-ghost" onClick={() => { setQ(""); void search(""); }}>Azzera</button>
   375→          </form>
   376→
   377→          {rows && rows.length > 0 ? (
   378→            <div className="grid mt2">
   379→              {rows.slice(0, 12).map(r => (
   380→                <article className="card card-click" key={r.entity.id} style={{ display: "flex", justifyContent: "space-between", alignItems: "center", gap: 12 }}>
   381→                  <div>
   382→                    <h3 style={{ margin: "0 0 4px" }}>{r.entity.name}</h3>

   386→                    </p>
   387→                  </div>

   390→              ))}
   391→              {rows.length > 12 ? <p className="small muted">Mostrati i primi 12 su {rows.length}: affina la ricerca per vedere gli altri.</p> : null}
   392→            </div>

   395→          ) : null}
   396→
   397→          <div className="section-head mt3">

   399→              <h2 style={{ marginBottom: 0 }}>Non c&apos;è ancora? Aggiungilo tu</h2>

   401→            </div>
   402→          </div>
   403→          <div className="card">
   404→            {!adding ? (
   405→              <button className="btn btn-primary" onClick={() => setAdding(true)}>+ Aggiungi un immobile da recensire</button>

   409→                if (address.trim().length < 5 || !zone) { setAddErr("Indica l'indirizzo (almeno 5 caratteri) e scegli la città dai suggerimenti del campo “Città, comune o paese”."); return; }
   410→                await addEntity();
   411→              }}>

   456→                      </ul>
   457→                    ) : <span className="hint">Nessun risultato: piazza il pin a mano con un clic sulla mappa (o col GPS).</span>

   465→                <div className="row" style={{ justifyContent: "flex-end" }}>
   466→                  <button type="button" className="btn btn-ghost" onClick={() => setAdding(false)}>Annulla</button>
   467→                  <button className="btn btn-primary" type="submit">Aggiungi e scrivi la recensione</button>
   468→                </div>
   469→              </form>
   470→            )}
   471→          </div>
   472→        </>
   473→      )}
   474→    </div>
   475→  );

   479→function OpenReviewForm({ sel, onBack, onDone }: {
   480→  sel: SelEntity;
   481→  onBack: () => void;
   482→  onDone: (entityId: string, premiumDays: number) => void | Promise<void>;
   483→}) {
   484→  const { boot, toast } = useApp();
   485→  const cat = boot?.categories.find(c => c.id === sel.categoryId) || null;
   486→  const [kind, setKind] = useState("");
   487→  const [form, setForm] = useState<FormState>({
   488→    kind: "", score: 5, crit: {} as Record<string, number>,
   489→    priceAmount: "", priceUnit: "", priceKind: "Concluso/pagato",
   490→    pros: "", cons: "", text: "", honesty: false, hp: "",
   491→  });
   492→  const [photos, setPhotos] = useState<string[]>([]);
   493→  const [err, setErr] = useState<string | null>(null);
   494→  const [busy, setBusy] = useState(false);
   495→
   496→  useEffect(() => { if (cat?.experienceKinds.length) setKind(cat.experienceKinds[0]); }, [cat]);
   497→
   498→  if (!boot || !cat) return <div className="container section"><p className="muted">Caricamento criteri…</p></div>;
   499→
   500→  return (
   501→    <div className="container" style={{ maxWidth: 760, paddingTop: 26, paddingBottom: 30 }}>
   502→      <div className="section-head">
   503→        <div>
   504→          <h1 style={{ margin: 0 }}>Scrivi la recensione</h1>
   505→          <p className="sub">
   506→            <ChipCat cat={cat} /> · {sel.name} · <span className="pill pill-gray">Esperienza dichiarata</span>
   507→          </p>
   508→        </div>
   509→        <button className="btn btn-ghost btn-sm" onClick={onBack}>← Cambia soggetto</button>
   510→      </div>
   511→      <Alert kind="info">
   512→        Stai scrivendo senza invito: la recensione sarà marcata <b>“esperienza dichiarata”</b> (non “verificata”). Restano valide le regole: segnalazioni, moderazione con decisioni motivate e revoca del Premium in caso di frode accertata.
   513→      </Alert>
   514→      {boot.site.premiumEnabled ? (
   515→        <Alert kind="ok"><b>Membership Premium.</b> Pubblicando questa recensione ottieni <b>{boot.settings.premiumDaysPerReview} giorni di Premium</b>: leggi le recensioni integrali di tutte le schede. E se alleghi una foto, sblocchi anche le foto di tutte le altre recensioni. Il premio è legato al contributo, non al voto.</Alert>
   516→      ) : null}

   520→          setErr(null); setBusy(true);
   521→          const res = await api<{ reviewId: string; entityId: string; photos?: { added: number; skipped: number }; premium: { grantedDays: number } | null }>("/api/reviews", {
   522→            body: { entityId: sel.id, experienceKind: kind, ...form, photos },
   523→          });
   524→          setBusy(false);
   525→          if (res.ok) {

   527→            await onDone(res.entityId, res.premium?.grantedDays || 0);
   528→          } else setErr(res.error);
   529→        }}>

   531→            <label htmlFor="rv-kind">Che esperienza hai avuto?</label>

   542→          </label>
   543→          <div className="hp-field"><label htmlFor="orv-hp">Non compilare</label><input id="orv-hp" tabIndex={-1} autoComplete="off" value={form.hp} onChange={e => setForm(f => ({ ...f, hp: e.target.value }))} /></div>
   544→          {err ? <p className="error-msg" style={{ display: "block" }}>{err}</p> : null}
   545→          <button className="btn btn-primary btn-block" type="submit" disabled={busy}>Pubblica la recensione</button>
   546→        </form>
   547→      </div>
   548→    </div>
   549→  );
