   165→   165→function FindStep({ target, onSelected, onBack }: {
   166→   166→  target: "immobile" | "zona";
   167→   167→  onSelected: (ent: SelEntity) => void;
   168→   168→  onBack: () => void;

   186→   186→  const [geoRes, setGeoRes] = useState<Array<{ label: string; lat: number; lng: number; zoneName: string | null; province: string | null; region: string | null; road?: string | null; houseNumber?: string | null }> | null>(null);
   187→   187→  const [geoBusy, setGeoBusy] = useState(false);
   188→   188→  const [gpsStage, setGpsStage] = useState<"idle" | "gps" | "addr">("idle");
   189→   189→  const [addErr, setAddErr] = useState<string | null>(null);

   191→   191→  // zona: elenco delle città già sul catalogo (dalla mappa pubblica)
   192→   192→  const [pubZones, setPubZones] = useState<PubZone[] | null>(null);
   193→   193→  const [zoneBusy, setZoneBusy] = useState(false);
   194→   194→
   195→   195→  // ogni via che imposta la città passa da qui: la conferma precedente

   197→   197→  const applyZone = (z: ZonePick | null) => {
   198→   198→    setZone(z);
   199→   199→    setZoneOk(false);

   201→   201→
   202→   202→  const search = useCallback(async (query: string) => {
   203→   203→    setBusy(true);
   204→   204→    const res = await api<{ rows: SearchRow[] }>(`/api/public/search?cat=immobile${query ? `&q=${encodeURIComponent(query)}` : ""}`);
   205→   205→    setRows(res.ok ? res.rows : []);
   206→   206→    setBusy(false);
   207→   207→  }, []);
   208→   208→
   209→   209→  useEffect(() => {
   210→   210→    if (target === "immobile") void search("");
   211→   211→    else void (async () => {
   212→   212→      const res = await api<{ zones: PubZone[] }>("/api/public/map");
   213→   213→      setPubZones(res.ok ? res.zones : []);

   218→   218→  // altrimenti nasce adesso (copertura nazionale, paesi inclusi)
   219→   219→  const activateZone = async (z: { id?: string; name: string; lat: number; lng: number; province?: string | null; region?: string | null; entityId?: string | null }) => {
   220→   220→    if (z.entityId) { onSelected({ id: z.entityId, name: z.name, categoryId: "zona" }); return; }
   221→   221→    setZoneBusy(true);
   222→   222→    const res = await api<{ entityId: string; name: string; created: boolean }>("/api/zones", {
   223→   223→      body: { name: z.name, province: z.province ?? null, region: z.region ?? null, lat: z.lat, lng: z.lng },
   224→   224→    });

   226→   226→    if (res.ok && res.entityId) {

   228→   228→      onSelected({ id: res.entityId, name: z.name, categoryId: "zona" });
   229→   229→    } else {
   230→   230→      toast("Non riesco ad attivare la scheda della città: riprova tra poco.", "err");
   231→   231→    }
   232→   232→  };
   233→   233→
   234→   234→  const doGeocode = async () => {
   235→   235→    if (geoQ.trim().length < 3) return;
   236→   236→    setGeoBusy(true); setGeoRes(null);
   237→   237→    const res = await api<{ results: Array<{ label: string; lat: number; lng: number; zoneName: string | null; province: string | null; region: string | null; road?: string | null; houseNumber?: string | null }> }>(`/api/public/geocode?q=${encodeURIComponent(geoQ.trim())}`);
   238→   238→    setGeoRes(res.ok ? res.results : []);
   239→   239→    setGeoBusy(false);
   240→   240→  };
   241→   241→
   242→   242→  // GPS + reverse geocoding: chi è davanti all'immobile preme un bottone e
   243→   243→  // il modulo si riempie da solo (indirizzo reale, città, pin). Le coordinate
   244→   244→  // vanno al geocoder pubblico SOLO in questo momento, su richiesta esplicita.
   245→   245→  const useGpsAddress = () => {
   246→   246→    if (!navigator.geolocation) { toast("Questo browser non espone la posizione: usa la ricerca o la mappa.", "err"); return; }

   249→   249→      async pos => {

   251→   251→        setGeo(pt);
   252→   252→        setGpsStage("addr");
   253→   253→        const res = await api<{ found: boolean; address: { road: string | null; houseNumber: string | null; zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${pt.lat}&lng=${pt.lng}`);
   254→   254→        setGpsStage("idle");
   255→   255→        if (res.ok && res.found && res.address) {
   256→   256→          const a = res.address;
   257→   257→          const street = [a.road, a.houseNumber].filter(Boolean).join(" ");
   258→   258→          if (street) setAddress(street);

   263→   263→        }
   264→   264→      },
   265→   265→      () => { setGpsStage("idle"); toast("Posizione non disponibile: consenti l'accesso dal browser oppure cerca a mano.", "err"); },
   266→   266→      { timeout: 9000, maximumAge: 30000, enableHighAccuracy: true },
   267→   267→    );
   268→   268→  };

   272→   272→  // cosa sta per salvare e lo conferma scegliendo il suggerimento
   273→   273→  const pickAddress = (p: AddressPlace) => {

   275→   275→    if (p.zoneName) {
   276→   276→      applyZone({ name: p.zoneName, province: p.province, region: p.region, lat: p.lat, lng: p.lng });
   277→   277→    }
   278→   278→  };

   280→   280→  // conferma di un risultato della ricerca sulla mappa: piazza il pin e

   282→   282→  // campagna mappata senza riferimento), lo chiede al reverse geocoder sul

   286→   286→  const pickGeoResult = async (g: NonNullable<typeof geoRes>[number]) => {

   290→   290→    if (street && !address.trim()) setAddress(street);

   292→   292→    if (g.zoneName) {

   294→   294→      return;
   295→   295→    }
   296→   296→    const res = await api<{ found: boolean; address: { zoneName: string | null; province: string | null; region: string | null } | null }>(`/api/public/geocode/reverse?lat=${g.lat}&lng=${g.lng}`);
   297→   297→    if (res.ok && res.found && res.address?.zoneName) {
   298→   298→      applyZone({ name: res.address.zoneName, province: res.address.province, region: res.address.region, lat: g.lat, lng: g.lng });
   299→   299→    } else {
   300→   300→      toast("Posizione segnata sulla mappa: scegli anche la città nel campo “Città, comune o paese”.", "err");
   301→   301→    }
   302→   302→  };
   303→   303→
   304→   304→  const addEntity = async () => {
   305→   305→    if (addBusy) return;
   306→   306→    setAddErr(null);
   307→   307→    setAddBusy(true);
   308→   308→    const res = await api<{ entity: { id: string; name: string } }>("/api/entities", {
   309→   309→      body: {
   310→   310→        address: address.trim(),
   311→   311→        // città già presente → id diretto; nuova località → nome+coordinate.
   312→   312→        // Le coordinate vanno solo se davvero valide (numeri finiti): il
   313→   313→        // server ha comunque i fallback (zona nota col nome, pin sulla mappa)
   314→   314→        zoneId: zone?.localId,
   315→   315→        zone: zone && !zone.localId
   316→   316→          ? {
   317→   317→            name: zone.name, province: zone.province, region: zone.region,
   318→   318→            lat: Number.isFinite(zone.lat) ? zone.lat : undefined,
   319→   319→            lng: Number.isFinite(zone.lng) ? zone.lng : undefined,
   320→   320→          }
   321→   321→          : undefined,
   322→   322→        geo,
   323→   323→      },
   324→   324→    });
   325→   325→    setAddBusy(false);
   326→   326→    if (res.ok) {
   327→   327→      if (res.entity) {
   328→   328→        toast("Immobile aggiunto: ora scrivi la recensione.");
   329→   329→        onSelected({ id: res.entity.id, name: res.entity.name, categoryId: "immobile" });
   330→   330→      }
   331→   331→      return;
   332→   332→    }
   333→   333→    const ex = (res as { existingId?: string }).existingId;
   334→   334→    if (ex) {
   335→   335→      toast("Questo immobile è già presente: puoi recensire la scheda esistente.");
   336→   336→      onSelected({ id: ex, name: address.trim(), categoryId: "immobile" });
   337→   337→      return;
   338→   338→    }
   339→   339→    setAddErr(res.error);
