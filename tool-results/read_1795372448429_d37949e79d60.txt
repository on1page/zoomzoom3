   174→   174→function Stepper({ labels, current, onJump }: { labels: string[]; current: number; onJump: (i: number) => void }) {

   176→   176→    <ol className="zz-stepper" aria-label="Avanzamento">

   182→   182→            aria-current={i === current ? "step" : undefined}
   183→   183→            onClick={i < current ? () => onJump(i) : undefined}
   184→   184→            onKeyDown={i < current ? (e => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); onJump(i); } }) : undefined}
   185→   185→            tabIndex={i < current ? 0 : undefined}
   186→   186→            title={i < current ? `Torna a: ${l}` : undefined}
   187→   187→          >
   188→   188→            <span className="zz-dot">{i < current ? <CheckSvg size={11} /> : i + 1}</span>

   190→   190→          </li>

   264→   264→function PlaceStep({ target, onBack, onInternal, onDraft }: {
   265→   265→  target: Target;
   266→   266→  onBack: () => void;
   267→   267→  onInternal: (ent: SelEntity, note: { created: boolean; reused: boolean }) => void;
   268→   268→  onDraft: (d: Draft) => void;
   269→   269→}) {
   270→   270→  const { boot, toast } = useApp();
   271→   271→  const [q, setQ] = useState("");
   272→   272→  const [internal, setInternal] = useState<SearchRow[] | null>(null);
   273→   273→  const [geoRes, setGeoRes] = useState<GeoRow[] | null>(null);
   274→   274→  const [busyInt, setBusyInt] = useState(false);
   275→   275→  const [busyGeo, setBusyGeo] = useState(false);
   276→   276→  const [throttled, setThrottled] = useState(false);
   277→   277→  const [gpsStage, setGpsStage] = useState<"idle" | "gps" | "addr">("idle");
   278→   278→  const [zoneBusy, setZoneBusy] = useState(false);
   279→   279→  const [pubZones, setPubZones] = useState<PubZone[] | null>(null);
   280→   280→  const [personBlock, setPersonBlock] = useState<SearchRow | null>(null);
   281→   281→
   282→   282→  const zoneName = useCallback((id: string | null) => boot?.zones.find(z => z.id === id)?.name || "", [boot]);
   283→   283→
   284→   284→  // zona: elenco delle città già sul catalogo (con la loro scheda, se attiva)
   285→   285→  useEffect(() => {
   286→   286→    if (target !== "zona") return;
   287→   287→    void (async () => {
   288→   288→      const res = await api<{ zones: PubZone[] }>("/api/public/map");
   289→   289→      setPubZones(res.ok ? res.zones : []);
   290→   290→    })();
   291→   291→  }, [target]);
   292→   292→
   293→   293→  // ricerca combinata: schede interne + luoghi della mappa
   294→   294→  const runSearch = useCallback(async (query: string) => {
   295→   295→    if (target !== "zona") {
   296→   296→      setBusyInt(true);
   297→   297→      const cat = target === "immobile" ? "immobile" : SUBJECT_CATS.join(",");
   298→   298→      const res = await api<{ rows: SearchRow[] }>(`/api/public/search?cat=${cat}${query ? `&q=${encodeURIComponent(query)}` : ""}`);
   299→   299→      setInternal(res.ok ? res.rows : []);
   300→   300→      setBusyInt(false);
   301→   301→    }
   302→   302→    if (target !== "altro" && query.length >= 3) {
   303→   303→      setBusyGeo(true);
   304→   304→      const res = await api<{ results: GeoRow[]; throttled?: boolean }>(`/api/public/geocode?q=${encodeURIComponent(query)}`);
   305→   305→      setThrottled(!!(res as { throttled?: boolean }).throttled);
   306→   306→      setGeoRes(res.ok ? res.results : []);
   307→   307→      setBusyGeo(false);
   308→   308→    } else {
   309→   309→      setGeoRes(null);
   310→   310→    }
   311→   311→  }, [target]);
   312→   312→
   313→   313→  // ricerca con debounce. A vuoto (immobile) si aprono le schede esistenti:
   314→   314→  // chi entra trova subito anche il catalogo, non solo la mappa (C5)
   315→   315→  useEffect(() => {
   316→   316→    const query = q.trim();
   317→   317→    if (query.length < 2) {
   318→   318→      setGeoRes(null); setThrottled(false);
   319→   319→      if (target === "immobile") void runSearch(""); else setInternal(null);
   320→   320→      return;
   321→   321→    }
   322→   322→    const t = setTimeout(() => { void runSearch(query); }, 420);
   323→   323→    return () => clearTimeout(t);
   324→   324→  }, [q, target, runSearch]);
   325→   325→
   326→   326→  // scheda interna già presente: la si usa così com'è (USE del diagramma)
   327→   327→  const pickInternal = (r: SearchRow) => {
   328→   328→    const cat = boot?.categories.find(c => c.id === r.entity.categoryId);
   329→   329→    if (cat?.isPerson) { setPersonBlock(r); return; }
   330→   330→    onInternal({ id: r.entity.id, name: r.entity.name, categoryId: r.entity.categoryId }, { created: false, reused: true });
   331→   331→  };
   332→   332→
   333→   333→  // città già sul catalogo: scheda attiva → si usa; altrimenti si attiva ora
   334→   334→  const pickZone = async (z: { id?: string; name: string; lat: number; lng: number; entityId?: string | null }) => {
   335→   335→    if (z.entityId) { onInternal({ id: z.entityId, name: z.name, categoryId: "zona" }, { created: false, reused: true }); return; }
   336→   336→    setZoneBusy(true);
   337→   337→    const res = await api<{ entityId: string; name: string; created: boolean }>("/api/zones", {
   338→   338→      body: { name: z.name, province: null, region: null, lat: z.lat, lng: z.lng },
   339→   339→    });
   340→   340→    setZoneBusy(false);
   341→   341→    if (res.ok && res.entityId) {
   342→   342→      toast(res.created ? `Scheda di ${z.name} attivata.` : `Scheda di ${z.name} già pronta.`);
   343→   343→      onInternal({ id: res.entityId, name: z.name, categoryId: "zona" }, { created: res.created, reused: !res.created });
   344→   344→    } else {
   345→   345→      toast("Non riesco ad attivare la scheda della città: riprova tra poco.", "err");
   346→   346→    }
   347→   347→  };
   348→   348→
   349→   349→  // luogo della mappa: indirizzo o città → minibozza che finisce al passo 3
   350→   350→  const pickGeo = (g: GeoRow) => {
   351→   351→    const street = [g.road, g.houseNumber].filter(Boolean).join(" ");

   635→   635→function ConfirmStep({ target, draft, setDraft, onBack, onReady }: {
   636→   636→  target: Target;
   637→   637→  draft: Draft;
   638→   638→  setDraft: (updater: (d: Draft) => Draft) => void;
   639→   639→  onBack: () => void;
   640→   640→  onReady: (ent: SelEntity, note: { created: boolean; reused: boolean }) => void;
   641→   641→}) {
   642→   642→  const { boot, toast } = useApp();
   643→   643→  const [editZone, setEditZone] = useState(!draft.zone);
   644→   644→  const [editAddr, setEditAddr] = useState(target === "immobile" && draft.address.trim().length < 5);
   645→   645→  const [showMap, setShowMap] = useState(false);
   646→   646→  const [busy, setBusy] = useState(false);
   647→   647→  const [err, setErr] = useState<string | null>(null);
   648→   648→  const [nearby, setNearby] = useState<NearbyRow[] | null>(null);
   649→   649→
   650→   650→  const isImmobile = target === "immobile";
   651→   651→  const addrOk = draft.address.trim().length >= 5;
   652→   652→  const valid = !!draft.zone && (!isImmobile || addrOk);
   653→   653→
   654→   654→  // conferma: proposta di riuso (C6) → creazione/dedup (EXIST/CREATE)
   655→   655→  const confirm = async (skipProx = false) => {
   656→   656→    if (busy || !valid) return;
   657→   657→    setErr(null); setBusy(true);
   658→   658→
   659→   659→    if (target === "zona") {
   660→   660→      const z = draft.zone!;
   661→   661→      const res = await api<{ entityId: string; created: boolean }>("/api/zones", {
   662→   662→        body: { name: z.name, province: z.province, region: z.region, lat: z.lat, lng: z.lng },
   663→   663→      });
   664→   664→      setBusy(false);
   665→   665→      if (res.ok && res.entityId) {
   666→   666→        toast(res.created ? `Scheda di ${z.name} attivata.` : `Scheda di ${z.name} già pronta.`);
   667→   667→        onReady({ id: res.entityId, name: z.name, categoryId: "zona" }, { created: res.created, reused: !res.created });
   668→   668→      } else setErr(res.error);
   669→   669→      return;
   670→   670→    }
   671→   671→
   672→   672→    // immobile — con il pin, prima propongo le schede vicine (entro 50 m)
   673→   673→    if (draft.geo && !skipProx) {
   674→   674→      const nr = await api<{ rows: NearbyRow[] }>(`/api/public/nearby?lat=${draft.geo.lat}&lng=${draft.geo.lng}${draft.zone?.localId ? `&zona=${draft.zone.localId}` : ""}`);
   675→   675→      if (nr.ok && nr.rows?.length) { setNearby(nr.rows); setBusy(false); return; }
   676→   676→    }
   677→   677→    setNearby(null);
   678→   678→
   679→   679→    const z = draft.zone;
   680→   680→    const res = await api<{ entity: { id: string; name: string } }>("/api/entities", {
   681→   681→      body: {
   682→   682→        address: draft.address.trim(),
   683→   683→        zoneId: z?.localId,
   684→   684→        zone: z && !z.localId
   685→   685→          ? { name: z.name, province: z.province, region: z.region, lat: z.lat, lng: z.lng }
   686→   686→          : undefined,
   687→   687→        geo: draft.geo ?? undefined,
   688→   688→      },
   689→   689→    });
   690→   690→    setBusy(false);
   691→   691→    if (res.ok && res.entity) {
   692→   692→      onReady({ id: res.entity.id, name: res.entity.name, categoryId: "immobile" }, { created: true, reused: false });
   693→   693→      return;
   694→   694→    }
   695→   695→    const ex = (res as { existingId?: string }).existingId;
