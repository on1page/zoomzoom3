    10→/* ---------- Invito ---------- */
    11→type InviteInfo =
    12→  | { state: "invalid" | "revoked" | "expired" | "pending"; message: string }
    13→  | {
    14→    state: "valid"; experience: { entityName: string; categoryId: string; kind: string; period: string | null };
    15→    maskedEmail: string; expiresAt: string; loggedIn: boolean; emailMatches: boolean;
    16→  };
    17→
    18→export function InviteView({ token }: { token: string }) {
    19→  const { user, refreshBoot, toast } = useApp();
    20→  const [info, setInfo] = useState<InviteInfo | null>(null);
    21→  const [busy, setBusy] = useState(false);
    22→
    23→  const load = useCallback(async () => {
    24→    const res = await api<InviteInfo>(`/api/invitations/info/${encodeURIComponent(token)}`);
    25→    setInfo(res.ok ? res as unknown as InviteInfo : { state: "invalid", message: "Il collegamento non è valido." });
    26→  }, [token]);
    27→  useEffect(() => { void load(); }, [load]);
    28→
    29→  if (!info) return <div className="container section" style={{ maxWidth: 640 }}><p className="muted">Verifica dell&apos;invito…</p></div>;
    30→
    31→  if (info.state !== "valid") {
    32→    return (
    33→      <div className="container section" style={{ maxWidth: 640 }}>
    34→        <h1 className="sr-only">Invito a recensire</h1>
    35→        <Alert kind="danger">
    36→          <h1 style={{ margin: "0 0 8px", fontSize: "1.3rem" }}>Invito non utilizzabile</h1>
    37→          {info.message}
    38→        </Alert>
    39→        {info.state === "expired" ? <p className="small">Se pensi sia un errore, scrivi dal modulo <a href="#/contatti">contatti</a>.</p> : null}
    40→        <p className="small mt3"><a href="#/">&#8592; Torna alla home</a></p>
    41→      </div>
    42→    );
    43→  }
    44→
    45→  const claim = async () => {
    46→    setBusy(true);
    47→    const res = await api<{ experienceId: string }>("/api/invitations/redeem", { body: { token } });
    48→    setBusy(false);
    49→    if (res.ok) {
    50→      await refreshBoot();
    51→      toast("Invito accettato: esperienza collegata al tuo account.");
    52→      navigate(`#/scrivi/${res.experienceId}`);
    53→    } else {
    54→      toast(res.error, "err");
    55→      void load();
    56→    }
    57→  };
    58→
    59→  return (
    60→    <div className="container section" style={{ maxWidth: 640 }}>
    61→      <h1 className="sr-only">Invito a recensire</h1>
    62→      <Alert kind="ok"><b>Invito valido</b> — collegato a un&apos;esperienza registrata su ZoomZoom.</Alert>
    63→      <div className="card mb2">
    64→        <h2 style={{ marginTop: 0, fontSize: "1.15rem" }}>Esperienza da recensire</h2>
    65→        <p className="mb1"><b>{info.experience.entityName}</b> · {info.experience.kind}{info.experience.period ? ` · ${info.experience.period}` : ""}</p>
    66→        <p className="small mb1">Destinatario: <b>{info.maskedEmail}</b> · Scadenza: {new Date(info.expiresAt).toLocaleString("it-IT")} · Uso singolo</p>
    67→        <p className="small muted mb0">La verifica attesta l&apos;esistenza dell&apos;interazione registrata (invito o documento): <b>non garantisce che ogni affermazione sia vera</b>. Le opinioni restano responsabilità di chi le scrive, con moderazione e appello.</p>
    68→      </div>
    69→      {!user ? (
    70→        <>
    71→          <Alert kind="info">Per usare l&apos;invito accedi o registrati con l&apos;indirizzo email a cui è destinato (<b>{info.maskedEmail}</b>): è così che colleghiamo la recensione all&apos;esperienza.</Alert>
    72→          <div className="row">
    73→            <a className="btn btn-primary" href="#/accedi">Accedi</a>
    74→            <a className="btn btn-ghost" href="#/registrati">Registrati con quell&apos;email</a>
    75→          </div>
    76→        </>
    77→      ) : !info.emailMatches ? (
    78→        <Alert kind="danger">Questo invito è destinato a un altro indirizzo email. Stai usando l&apos;account <b>{user.email}</b>.</Alert>
    79→      ) : (
    80→        <button className="btn btn-primary" onClick={claim} disabled={busy}>Accetta l&apos;invito e scrivi la recensione</button>
    81→      )}
    82→      <p className="small mt3"><a href="#/">&#8592; Torna alla home</a></p>
    83→    </div>
    84→  );
    85→}

    92→  const [existingReviewId, setExistingReviewId] = useState<string | null>(null);
    93→
    94→  useEffect(() => {
    95→    if (!user) return;
    96→    void api<{ experiences: Array<{ id: string; entity: { id: string; name: string }; kind: string; period: string | null; status: string; review: { id: string } | null }> }>("/api/me/overview").then(res => {
    97→      if (!res.ok) return;

   256→/* ---------- Rettifica ---------- */
   257→export function UpdateReviewView({ reviewId }: { reviewId: string }) {
   258→  const { user, boot, toast } = useApp();
   259→  const [state, setState] = useState<"loading" | "form" | "notFound" | "notEditable" | "windowOver">("loading");
   260→  const [rev, setRev] = useState<{
   261→    entityId: string; entityName: string; catId: string; score: number;
   262→    criteria: Record<string, number>; pros: string[]; cons: string[]; text: string;
   263→    price: { amount: number; unit: string; kind: string } | null; createdAt: string;
   264→  } | null>(null);
   265→
   266→  useEffect(() => {
   267→    if (!user) return;
   268→    void (async () => {
   269→      const res = await api<{ reviews: Array<{ id: string; entity: { id: string; name: string }; score: number; status: string; createdAt: string }> }>("/api/me/overview");
   270→      if (!res.ok) return;
   271→      const r = res.reviews.find(x => x.id === reviewId);
   272→      if (!r) { setState("notFound"); return; }
   273→      if (r.status !== "published") { setState("notEditable"); return; }
   274→      const days = (Date.now() - new Date(r.createdAt).getTime()) / 86400000;
   275→      if (days > (boot?.settings.rettificaDays ?? 30)) { setState("windowOver"); return; }
   276→      const full = await api<{ reviews: Array<{ id: string; criteria: Record<string, number>; pros: string[]; cons: string[]; text: string; price: { amount: number; unit: string; kind: string } | null; entity: { id: string; name: string } }> }>(`/api/public/entity/${r.entity.id}`);
   277→      if (full.ok) {
   278→        const fr = (full as unknown as { reviews: Array<{ id: string; criteria: Record<string, number>; pros: string[]; cons: string[]; text: string; price: { amount: number; unit: string; kind: string } | null }> }).reviews.find(x => x.id === reviewId);
   279→        if (fr) {
   280→          setRev({
   281→            entityId: r.entity.id, entityName: r.entity.name, catId: "", score: r.score,
   282→            criteria: fr.criteria, pros: fr.pros, cons: fr.cons, text: fr.text, price: fr.price, createdAt: r.createdAt,
   283→          });
   284→          setState("form");
   285→          return;
   286→        }
   287→      }
   288→      setState("notFound");
   289→    })();
   290→  }, [user, reviewId, boot]);
   291→
   292→  if (!user) return <div className="container section" style={{ maxWidth: 620 }}><Alert kind="info">Serve <a href="#/accedi">accedere</a> per rettificare una recensione.</Alert></div>;
   293→  if (state === "loading") return <div className="container section"><p className="muted">Caricamento…</p></div>;
   294→  if (state === "notFound") return <div className="container section" style={{ maxWidth: 620 }}><Alert kind="danger">Recensione non trovata.</Alert></div>;
   295→  if (state === "notEditable") return (
   296→    <div className="container section" style={{ maxWidth: 620 }}>
   297→      <Alert kind="danger">La recensione non è modificabile in autonomia: è stata nascosta dalla moderazione. Scrivi dal modulo <a href="#/contatti">contatti</a> per la procedura di correzione.</Alert>
   298→    </div>
   299→  );
   300→  if (state === "windowOver" || !rev || !boot) return (
   301→    <div className="container section" style={{ maxWidth: 620 }}>
   302→      <Alert kind="warn">
   303→        <h1 style={{ margin: "0 0 8px", fontSize: "1.3rem" }}>Finestra di rettifica conclusa</h1>
   304→        Puoi aggiornare autonomamente entro {boot?.settings.rettificaDays ?? 30} giorni dalla pubblicazione. Dopo, le correzioni passano dalla moderazione: usa il modulo <a href="#/contatti">contatti</a> indicando la recensione.
   305→      </Alert>
   306→    </div>
   307→  );
   308→
   309→  return <UpdateReviewForm reviewId={reviewId} rev={rev} />;
   310→}
